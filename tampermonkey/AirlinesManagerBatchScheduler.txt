/* ============================================================================
 * AIRLINES MANAGER BATCH SCHEDULER v1.0.0 — ANNOTATED REFERENCE
 * ============================================================================
 * PURPOSE
 *   A Tampermonkey userscript that runs on Airlines Manager planning pages.
 *   It accepts multiple weekly rotation requests, potentially spanning several
 *   hubs and aircraft types, and prepares schedules for seven empty aircraft
 *   per requested rotation. It provides read-only generation and live preflight,
 *   with a separately gated path for deploying schedules and aircraft names.
 *
 * HIGH-LEVEL WORKFLOW
 *   1. Discover planning JSON URLs and associate them with actual hub IDs/codes.
 *   2. Read each required hub's routes, aircraft and model specifications.
 *   3. Parse HUB | AIRCRAFT TYPE | ROTATION NAME | DESTINATIONS requests.
 *   4. Resolve routes, check range/category and reserve seven unused aircraft.
 *   5. Construct one weekly route pattern, then stagger seven aircraft by day.
 *   6. Validate generated schedules and show a read-only batch preview.
 *   7. On request, perform fresh live preflight; only an unchanged passing
 *      batch can reach the live deployment path.
 *   8. During deployment, submit schedules and renames with verification and
 *      audit logging. LIVE DEPLOYMENT CHANGES THE GAME ACCOUNT.
 *
 * IMPORTANT CONCEPTS
 *   - hubRegistry: map of hub airport codes to live planning data and catalogues.
 *   - route key: normalised HUB/DESTINATION string.
 *   - generatedBatch: generated schedules and permanently identified aircraft.
 *   - batchPreflightState: safety state used to gate live deployment.
 *   - one rotation: seven separate aircraft, offset by one weekday each.
 *   - times: seconds within a repeating 168-hour week.
 *
 * LIMITATIONS / CAUTIONS
 *   - This documentation describes the source as written; it is not a claim
 *     that the script has passed an end-to-end live test.
 *   - Route names must match the parser's HUB / DESTINATION convention.
 *   - The input parser currently accepts FOUR pipe-separated fields, despite
 *     an older nearby comment mentioning a six-field optimiser format.
 *   - Aircraft eligibility requires an empty planningList and 0 utilisation.
 *   - The game may change its response fields or write endpoints over time.
 *
 * DOCUMENTATION-ONLY EDIT: all original executable code is preserved.
 * ============================================================================ */

  /**
   * normaliseCode()
   * Standardises airport codes by trimming whitespace and converting to uppercase.

  function normaliseCode(value) {

    return String(

  /**
   * moduloWeek()
   * Wraps a timestamp into the repeating 604,800-second scheduling week, including negative offsets.
   *
   * Notes: This describes the existing behaviour; no logic is changed.
   */
  function moduloWeek(seconds) {

    const value =
      Number(seconds);


    return (
      (
        value %
        WEEK_SECONDS
      ) +
      WEEK_SECONDS
    ) %
    WEEK_SECONDS;
  }

  /**
   * secondsToDuration()
   * Formats a duration in seconds as hours and minutes for display.
   *
   * Notes: This describes the existing behaviour; no logic is changed.
   */
  function secondsToDuration(seconds) {

    const total =
      Math.round(
        Number(seconds)
      );


    if (
      !Number.isFinite(
        total
      )
    ) {

      return "—";
    }


    const hours =
      Math.floor(
        total /
        3600
      );


    const minutes =
      Math.floor(
        (
          total %
          3600
        ) /
        60
      );


    if (
      minutes ===
      0
    ) {

      return `${hours}h`;
    }


    return (
      `${hours}h ` +
      `${minutes}m`
    );
  }


  /**
   * secondsToWeekTime()
   * Converts a cyclic weekly timestamp to a weekday and HH:MM label.
   *
   * Notes: This describes the existing behaviour; no logic is changed.
   */
  function secondsToWeekTime(seconds) {

    const value =
      moduloWeek(
        seconds
      );


    const dayIndex =
      Math.floor(
        value /
        DAY_SECONDS
      );


    const secondsIntoDay =
      value %
      DAY_SECONDS;


    const hour =
      Math.floor(
        secondsIntoDay /
        3600
      );


    const minute =
      Math.floor(
        (
          secondsIntoDay %
          3600
        ) /
        60
      );


    return (
      `${SHORT_DAYS[dayIndex]} ` +
      `${String(hour).padStart(2, "0")}:` +
      `${String(minute).padStart(2, "0")}`
    );
  }

/**
 * isAircraftEmpty()
 * Accepts aircraft only when their planning list is empty and reported utilisation is exactly zero.
 *
 * Notes: This describes the existing behaviour; no logic is changed.
 */
function isAircraftEmpty(
  aircraft
) {

  if (
    !aircraft ||
    !Array.isArray(
      aircraft.planningList
    ) ||
    aircraft.utilizationPercentage == null
  ) {

    return false;
  }


  const utilization =
    Number(
      aircraft.utilizationPercentage
    );


  return (
    aircraft.planningList.length === 0 &&
    Number.isFinite(
      utilization
    ) &&
    utilization === 0
  );
}

  /**
   * getAircraftArray()
   * Finds the aircraft array in alternative planning-JSON layouts, with a structural fallback.
   *
   * Notes: This describes the existing behaviour; no logic is changed.
   */
  function getAircraftArray(
    data
  ) {

    /*
     * Preserve support for the structures observed in the
     * Airlines Manager planning JSON.
     */

    const candidates = [
      data?.aircraftList,
      data?.aircraftDataArray,
      data?.aircrafts,
      data?.aircraft
    ];


    for (
      const candidate of
      candidates
    ) {

      if (
        Array.isArray(
          candidate
        )
      ) {

        return candidate;
      }
    }


    /*
     * Last-resort discovery for AM responses whose property
     * name changes but still contain the familiar aircraft
     * objects.
     */

    for (
      const value of
      Object.values(
        data || {}
      )
    ) {

      if (
        !Array.isArray(
          value
        ) ||
        value.length ===
          0
      ) {

        continue;
      }


      const sample =
        value[0];


      if (
        sample &&
        typeof sample ===
          "object" &&
        "aircraftListName" in
          sample &&
        "hubId" in
          sample &&
        "planningList" in
          sample
      ) {

        return value;
      }
    }


    return [];
  }


  /**
   * getRouteArray()

  function getRouteArray(

    const candidates = [

      data?.lineList

    for (

      if (
        Array.isArray(

        return candidate;

    }


    return [];
  }

  /**
   * buildAircraftTypeCatalogue()
   * Derives unique aircraft model specifications from live aircraft; rejects contradictory specifications.
   *
   * Notes: This describes the existing behaviour; no logic is changed.
   */
  function buildAircraftTypeCatalogue(
    data
  ) {

    const catalogue =
      new Map();


    const aircraft =
      getAircraftArray(
        data
      );


    for (
      const plane of
      aircraft
    ) {

      const type =
        String(
          plane?.aircraftListName ??
          ""
        ).trim();


      if (!type) {

        continue;
      }


      const spec = {

        type,

        speed:
          Number(
            plane.speed
          ),

        range:
          Number(
            plane.range
          ),

        category:
          Number(
            plane.category
          )
      };


      if (
        !Number.isFinite(
          spec.speed
        ) ||
        !Number.isFinite(
          spec.range
        ) ||
        !Number.isFinite(
          spec.category
        )
      ) {

        continue;
      }


      if (
        catalogue.has(
          type
        )
      ) {

        const existing =
          catalogue.get(
            type
          );


        if (
          Number(
            existing.speed
          ) !==
            spec.speed ||
          Number(
            existing.range
          ) !==
            spec.range ||
          Number(
            existing.category
          ) !==
            spec.category
        ) {

          throw new Error(
            `Airlines Manager returned inconsistent specifications ` +
            `for aircraft type ${type}.`
          );
        }


        continue;
      }


      catalogue.set(
        type,
        spec
      );
    }


    return catalogue;
  }


  /**
   * getAircraftTypeSpec()
   * Retrieves a model's speed, range and category; fails if unavailable.
   *
   * Notes: This describes the existing behaviour; no logic is changed.
   */
  function getAircraftTypeSpec(
    aircraftType,
    typeCatalogue
  ) {

    const catalogue =
      typeCatalogue;


    if (
      !catalogue ||
      typeof catalogue.get !==
        "function"
    ) {

      throw new Error(
        "Aircraft type catalogue is unavailable."
      );
    }


    const spec =
      catalogue.get(
        aircraftType
      );


    if (!spec) {

      throw new Error(
        `Aircraft type ${aircraftType} was not found ` +
        `in the live planning data for this hub.`
      );
    }


    return spec;
  }


  /* ============================================================
     DURATION FORMULA

     AM fallback formula:

       ((distance / speed) + 1) × 2

     rounded UP to the next 0.25 hour.

     Authoritative same-type lineList observations override
     this later when available.
  ============================================================ */

  /**
   * calculateRotationDuration()
   * Estimates round-trip time from distance and speed, including overhead, rounded up to 15 minutes.
   *
   * Notes: This describes the existing behaviour; no logic is changed.
   */
  function calculateRotationDuration(
    distance,
    speed
  ) {

    const distanceKm =
      Number(
        distance
      );


    const speedKmh =
      Number(
        speed
      );


    if (
      !Number.isFinite(
        distanceKm
      ) ||
      distanceKm <=
        0
    ) {

      throw new Error(
        "Cannot calculate duration: invalid route distance."
      );
    }


    if (
      !Number.isFinite(
        speedKmh
      ) ||
      speedKmh <=
        0
    ) {

      throw new Error(
        "Cannot calculate duration: invalid aircraft speed."
      );
    }


    const rawHours =
      (
        (
          distanceKm /
          speedKmh
        ) +
        1
      ) *
      2;


    const roundedQuarterHours =
      Math.ceil(
        rawHours *
        4
      ) /
      4;


    return (
      roundedQuarterHours *
      3600
    );
  }


  /* ============================================================

  /**
   * addRouteToCatalogue()
   * Parses a route name into hub and destination and stores its ID, distance and category; prefers primary route data.
   *
   * Notes: This describes the existing behaviour; no logic is changed.
   */
  function addRouteToCatalogue(
    routeMap,
    route,
    source
  ) {

    if (!route) {

      return;
    }


    const name =
      String(
        route.name ??
        ""
      );


    const match =
      name.match(
        /^\s*([A-Z0-9]{3,4})\s*\/\s*([A-Z0-9]{3,4})\s*$/i
      );


    if (!match) {

      return;
    }


    const hub =
      normaliseCode(
        match[1]
      );


    const destination =
      normaliseCode(
        match[2]
      );


    const key =
      routeKey(
        hub,
        destination
      );


    const existing =
      routeMap.get(
        key
      );


    const candidate = {

      ...route,

      id:
        Number(
          route.id
        ),

      name:
        `${hub} / ${destination}`,

      hub,

      destination,

      distance:
        Number(
          route.distance
        ),

      category:
        Number(
          route.category
        ),

      source
    };


    /*
     * lineDataArray is preferred because it contains the
     * live route category required for runway compatibility.
     */

    if (
      !existing ||
      source ===
        "lineDataArray"
    ) {

      routeMap.set(
        key,
        candidate
      );
    }
  }


  /**
   * buildRouteCatalogues()
   * Combines primary routes with aircraft lineList observations, retaining observed durations for matching aircraft types.
   *
   * Notes: This describes the existing behaviour; no logic is changed.
   */
  function buildRouteCatalogues(
    data
  ) {

    const routes =
      new Map();


    const observations =
      [];


    /*
     * Primary route catalogue.
     *
     * This supplies:
     *   line ID
     *   distance
     *   airport category
     */

    for (
      const route of
      getRouteArray(
        data
      )
    ) {

      addRouteToCatalogue(
        routes,
        route,
        "lineDataArray"
      );
    }


    /*
     * Aircraft lineList entries provide authoritative
     * same-aircraft-type rotation durations.
     */

    for (
      const aircraft of
      getAircraftArray(
        data
      )
    ) {

      const lineList =
        aircraft?.lineList;


      if (
        !lineList ||
        typeof lineList !==
          "object"
      ) {

        continue;
      }


      for (
        const [
          lineId,
          line
        ] of
        Object.entries(
          lineList
        )
      ) {

        addRouteToCatalogue(
          routes,
          {
            ...line,

            id:
              Number(
                lineId
              )
          },
          "lineList"
        );


        observations.push({

          id:
            Number(
              lineId
            ),

          lineId:
            Number(
              lineId
            ),

          name:
            line.name,

          distance:
            Number(
              line.distance
            ),

          duration:
            Number(
              line.duration
            ),

          aircraftType:
            aircraft.aircraftListName,

          aircraftId:
            Number(
              aircraft.id
            )
        });
      }
    }


    return {

      routes,

      observations
    };
  }


  /**
   * findRouteInMap()
   * Looks up one hub/destination pair in a route catalogue, returning null if absent.
   *
   * Notes: This describes the existing behaviour; no logic is changed.
   */
  function findRouteInMap(
    routeMap,
    hub,
    destination
  ) {

    if (
      !routeMap ||
      typeof routeMap.get !==
        "function"
    ) {

      return null;
    }


    return (
      routeMap.get(
        routeKey(
          hub,
          destination
        )
      ) ||
      null
    );
  }

  /**
   * resolveRoute()
   * Resolves a route and model specification; checks route ID, distance and category; prefers matching observed duration.
   *
   * Notes: This describes the existing behaviour; no logic is changed.
   */
  function resolveRoute(
    catalogues,
    aircraftType,
    hub,
    destination,
    typeCatalogue
  ) {

    const hubCode =
      normaliseCode(
        hub
      );


    const destinationCode =
      normaliseCode(
        destination
      );


    const route =
      findRouteInMap(
        catalogues?.routes,
        hubCode,
        destinationCode
      );


    if (!route) {

      return null;
    }


        const routeId =
  Number(
    route.id
  );

if (
  !Number.isSafeInteger(
    routeId
  ) ||
  routeId <= 0
) {

  throw new Error(
    `${hubCode}/${destinationCode} has an invalid route ID.`
  );
}

    const spec =
      getAircraftTypeSpec(
        aircraftType,
        typeCatalogue
      );


    const distance =
      Number(
        route.distance
      );


    if (
      !Number.isFinite(
        distance
      ) ||
      distance <=
        0
    ) {

throw new Error(
  `${hubCode}/${destinationCode} has an invalid distance.`
);
    }


    const routeCategory =
      Number(
        route.category
      );


    const aircraftCategory =
      Number(
        spec.category
      );


    if (
      !Number.isFinite(
        routeCategory
      )
    ) {

throw new Error(
  `${hubCode}/${destinationCode} has no valid airport category ` +
  `in the live Airlines Manager route data.`
);
    }


    if (
      !Number.isFinite(
        aircraftCategory
      )
    ) {

      throw new Error(
        `${aircraftType} has no valid aircraft category.`
      );
    }


    let duration =
      calculateRotationDuration(
        distance,
        Number(
          spec.speed
        )
      );


    let durationSource =
      "calculated";


    /*
     * Prefer an authoritative AM duration from another
     * aircraft of exactly the same type operating exactly
     * the same line.
     */

    const matches =
      (
        Array.isArray(
          catalogues?.observations
        )
          ? catalogues.observations
          : []
      )
        .filter(
          observation =>

            String(
              observation.aircraftType ??
              ""
            ) ===
              String(
                aircraftType
              ) &&

Number(
  observation.lineId ??
  observation.id
) ===
  routeId &&

            Number.isSafeInteger(
              Number(
                observation.duration
              )
            ) &&

            Number(
              observation.duration
            ) >
              0
        );


    const observedDurations =
      [
        ...new Set(
          matches.map(
            observation =>
              Number(
                observation.duration
              )
          )
        )
      ];


    if (
      observedDurations.length >
      1
    ) {

      throw new Error(
        `${hubCode}/${destinationCode} has conflicting ` +
        `${aircraftType} durations in live Airlines Manager data: ` +
        observedDurations
          .map(
            secondsToDuration
          )
          .join(", ")
      );
    }


    if (
      observedDurations.length ===
      1
    ) {

      duration =
        observedDurations[0];


      durationSource =
        "AM lineList";
    }


return {

  ...route,

  id:
    routeId,
      name:
        route.name ||
        `${hubCode} / ${destinationCode}`,

      hub:
        hubCode,

      destination:
        destinationCode,

      distance,

      category:
        routeCategory,

      aircraftCategory,

      categoryCompatible:
        routeCategory >=
        aircraftCategory,

      duration:
        Number(
          duration
        ),

      durationSource,

      aircraftType,

      aircraftSpeed:
        Number(
          spec.speed
        ),

      aircraftRange:
        Number(
          spec.range
        ),

      lookupSource:
        route.source
    };
  }


  /* ============================================================
     RANGE + AIRPORT CATEGORY VALIDATION
  ============================================================ */

  /**
   * validateRange()
   * Rejects routes beyond the aircraft's range or below its required airport category.
   *
   * Notes: This describes the existing behaviour; no logic is changed.
   */
  function validateRange(
    route,
    aircraftSpec
  ) {

    if (!route) {

      throw new Error(
        "Cannot validate an empty route."
      );
    }


    if (!aircraftSpec) {

      throw new Error(
        `${route.name || "Route"} has no aircraft specification.`
      );
    }


    const distance =
      Number(
        route.distance
      );


    const range =
      Number(
        aircraftSpec.range
      );


    if (
      !Number.isFinite(
        distance
      ) ||
      distance <=
        0
    ) {

      throw new Error(
        `${route.name || "Route"} has no valid distance.`
      );
    }


    if (
      !Number.isFinite(
        range
      ) ||
      range <=
        0
    ) {

      throw new Error(
        `${aircraftSpec.type || route.aircraftType || "Aircraft"} ` +
        `has no valid range.`
      );
    }


    if (
      distance >
      range
    ) {

      throw new Error(
        `${route.name} is ${distance.toLocaleString()} km, ` +
        `which exceeds the ` +
        `${aircraftSpec.type || route.aircraftType} range of ` +
        `${range.toLocaleString()} km.`
      );
    }


    const routeCategory =
      Number(
        route.category
      );


    const aircraftCategory =
      Number(
        aircraftSpec.category
      );


    if (
      !Number.isFinite(
        routeCategory
      )
    ) {

      throw new Error(
        `${route.name} has no valid airport category.`
      );
    }


    if (
      !Number.isFinite(
        aircraftCategory
      )
    ) {

      throw new Error(
        `${aircraftSpec.type || route.aircraftType || "Aircraft"} ` +
        `has no valid aircraft category.`
      );
    }


    if (
      routeCategory <
      aircraftCategory
    ) {

      throw new Error(
        `${route.name} is airport category ${routeCategory}, ` +
        `but ${aircraftSpec.type || route.aircraftType} requires ` +
        `category ${aircraftCategory} or higher.`
      );
    }


    return true;
  }


  /* ============================================================
     BATCH INPUT FORMAT

     Two formats are supported.

     Existing / legacy format:

       HUB | AIRCRAFT TYPE | ROTATION NAME | DEST1,DEST2,...

     Optimiser format:

       HUB | AIRCRAFT TYPE | ECO | CARGO | ROTATION NAME | DEST1,DEST2,...

     Business and First are currently fixed at zero in the
     optimiser format.

     Examples:

       BOM | A380-800 | BOM-C110 | NIM,LBV,YIA,SZX

     Repeated destinations are deliberately permitted.

     Lines beginning with # are ignored.
  ============================================================ */

  /**
   * parseBatchInput()
   * Parses pipe-delimited rotation requests; normalises fields and rejects missing values, hub-as-destination and duplicate names.
   *
   * Notes: This describes the existing behaviour; no logic is changed.
   */
  function parseBatchInput(
    raw
  ) {

    const lines =
      String(
        raw ?? ""
      )
        .split(
          /\r?\n/
        )
        .map(
          line =>
            line.trim()
        )
        .filter(
          line =>
            line &&
            !line.startsWith(
              "#"
            )
        );


    if (
      lines.length ===
      0
    ) {

      throw new Error(
        "Enter at least one rotation."
      );
    }


    const rotations =
      [];


    const seenRotationNames =
      new Set();


    for (
      let index = 0;
      index < lines.length;
      index++
    ) {

      const lineNumber =
        index + 1;


      const parts =
        lines[index]
          .split(
            "|"
          )
          .map(
            part =>
              part.trim()
          );


if (
  parts.length !== 4
) {

  throw new Error(
    `Line ${lineNumber}: expected ` +
    `HUB | AIRCRAFT TYPE | ROTATION NAME | DEST1,DEST2,...`
  );
}

      const hub =
        normaliseCode(
          parts[0]
        );


      const aircraftType =
        normaliseAircraftType(
          parts[1]
        );


      // Preserve existing aircraft seating and cargo.
      const requiredConfig =
        null;


      const rotationName =
        normaliseRotationName(
          parts[2]
        );


      const destinationText =
        parts[3];

      const destinations =
        destinationText
          .split(
            ","
          )
          .map(
            normaliseCode
          )
          .filter(
            Boolean
          );


      if (!hub) {

        throw new Error(
          `Line ${lineNumber}: hub is missing.`
        );
      }


      if (!aircraftType) {

        throw new Error(
          `Line ${lineNumber}: aircraft type is missing.`
        );
      }


      if (!rotationName) {

        throw new Error(
          `Line ${lineNumber}: rotation name is missing.`
        );
      }


      if (
        destinations.length ===
        0
      ) {

        throw new Error(
          `Line ${lineNumber}: no destinations were supplied.`
        );
      }


      if (
        destinations.some(
          destination =>
            destination ===
            hub
        )
      ) {

        throw new Error(
          `Line ${lineNumber}: do not include hub ${hub} ` +
          `in the destination list.`
        );
      }


      if (
        seenRotationNames.has(
          rotationName
        )
      ) {

        throw new Error(
          `Rotation name ${rotationName} is used more than once ` +
          `in this batch.`
        );
      }


      seenRotationNames.add(
        rotationName
      );


      rotations.push({

        batchIndex:
          rotations.length,

        sourceLine:
          lineNumber,

        hub,

        aircraftType,

        requiredConfig,

        rotationName,

        destinations
      });
    }


    return rotations;
  }

  /* ============================================================
     END CHUNK 1
  ============================================================ */

  /* ============================================================
     MULTI-HUB PLANNING DATA DISCOVERY

     Airlines Manager loads one planning JSON dataset per hub:

       /network/planning/load/{networkId}

     The numeric value in that URL is NOT the hub airport ID.

     The JSON response itself contains:

       hubAirportId

     v1.0 discovers every planning dataset currently exposed
     by the Planning page and builds a registry keyed by the
     actual hub airport ID.

     We do not hard-code BKK, BOM, or any other hub.
  ============================================================ */


  /**
   * getPlanningResourceUrls()
   * Reads browser performance entries to discover planning-data URLs requested by the page.
   *
   * Notes: This describes the existing behaviour; no logic is changed.
   */
  function getPlanningResourceUrls() {

    const resources =
      performance.getEntriesByType(
        "resource"
      );


    const urls =
      resources
        .map(
          entry =>
            entry.name
        )
        .filter(
          url =>
            /\/network\/planning\/load\/\d+(?:[?#].*)?$/.test(
              url
            )
        );


    return [
      ...new Set(
        urls
      )
    ];
  }


  /* ============================================================
     FETCH ONE PLANNING DATASET

     expectedHubId is optional.

     During discovery we do not yet know which hub a URL
     represents, so it is omitted.

     During later fresh validation we DO know the hub ID and
     require the fresh response to match it.
  ============================================================ */

  /**
   * fetchPlanningUrl()
   * Fetches authenticated live planning JSON without cache; validates HTTP, JSON and expected hub ID.
   *
   * Notes: This describes the existing behaviour; no logic is changed.
   */
  async function fetchPlanningUrl(
    url,
    expectedHubId = null
  ) {

    const response =
      await fetch(
        url,
        {
          method:
            "GET",

          credentials:
            "include",

          headers: {

            "Accept":
              "application/json, text/javascript, */*; q=0.01",

            "X-Requested-With":
              "XMLHttpRequest"
          },

          cache:
            "no-store"
        }
      );


    if (!response.ok) {

      throw new Error(
        `Planning GET failed: HTTP ${response.status}`
      );
    }


    let data;


    try {

      data =
        await response.json();

    } catch (error) {

      throw new Error(
        `Planning GET returned invalid JSON for ${url}.`
      );
    }


    const actualHubId =
      Number(
        data?.hubAirportId
      );


    if (
      !Number.isFinite(
        actualHubId
      )
    ) {

      throw new Error(
        `Planning dataset ${url} has no valid hubAirportId.`
      );
    }


    if (
      expectedHubId !==
        null &&
      Number(
        expectedHubId
      ) !==
        actualHubId
    ) {

      throw new Error(
        `Planning data belongs to hub ID ${actualHubId}, ` +
        `not expected hub ID ${expectedHubId}.`
      );
    }


    return data;
  }


  /* ============================================================
     DISCOVER HUB CODE FROM ROUTE DATA

     The planning JSON gives us hubAirportId but we also need
     the airport code used by the user, e.g.:

       49 -> BKK
       83 -> BOM

     We derive it from route names in lineDataArray / lineList.

     We deliberately require a consistent answer rather than
     guessing.
  ============================================================ */

  /**
   * discoverHubCodeFromData()
   * Infers a hub's airport code from route names and airport IDs, falling back to lineList names.
   *
   * Notes: This describes the existing behaviour; no logic is changed.
   */
  function discoverHubCodeFromData(
    data
  ) {

    const candidates =
      new Map();


    /**
     * addCandidate()
     * Counts one normalised airport-code candidate during hub-code inference.
     *
     * Notes: This describes the existing behaviour; no logic is changed.
     */
    function addCandidate(
      code
    ) {

      const normalised =
        normaliseCode(
          code
        );


      if (!normalised) {

        return;
      }


      candidates.set(
        normalised,
        (
          candidates.get(
            normalised
          ) ||
          0
        ) +
        1
      );
    }


    /*
     * Primary source: lineDataArray.
     *
     * airportOneId should normally equal hubAirportId.
     */

    for (
      const route of
      getRouteArray(
        data
      )
    ) {

      const name =
        String(
          route?.name ??
          ""
        );


      const match =
        name.match(
          /^\s*([A-Z0-9]{3,4})\s*\/\s*([A-Z0-9]{3,4})\s*$/i
        );


      if (!match) {

        continue;
      }


      const first =
        normaliseCode(
          match[1]
        );


      const second =
        normaliseCode(
          match[2]
        );


      const airportOneId =
        Number(
          route.airportOneId
        );


      const airportTwoId =
        Number(
          route.airportTwoId
        );


      const hubId =
        Number(
          data?.hubAirportId
        );


      if (
        Number.isFinite(
          airportOneId
        ) &&
        airportOneId ===
          hubId
      ) {

        addCandidate(
          first
        );


        continue;
      }


      if (
        Number.isFinite(
          airportTwoId
        ) &&
        airportTwoId ===
          hubId
      ) {

        addCandidate(
          second
        );


        continue;
      }


      /*
       * Fallback for route objects without usable airport IDs.
       *
       * In the planning datasets we have observed, the hub is
       * normally the first code.
       */

      addCandidate(
        first
      );
    }


    /*
     * Secondary source: aircraft lineList.
     *
     * This is useful if lineDataArray is unexpectedly sparse.
     */

    if (
      candidates.size ===
      0
    ) {

      for (
        const aircraft of
        getAircraftArray(
          data
        )
      ) {

        const lineList =
          aircraft?.lineList;


        if (
          !lineList ||
          typeof lineList !==
            "object"
        ) {

          continue;
        }


        for (
          const line of
          Object.values(
            lineList
          )
        ) {

          const name =
            String(
              line?.name ??
              ""
            );


          const match =
            name.match(
              /^\s*([A-Z0-9]{3,4})\s*\/\s*([A-Z0-9]{3,4})\s*$/i
            );


          if (!match) {

            continue;
          }


          addCandidate(
            match[1]
          );
        }
      }
    }


    if (
      candidates.size ===
      0
    ) {

      throw new Error(
        `Could not determine the airport code for hub ID ` +
        `${data?.hubAirportId ?? "unknown"}.`
      );
    }


    const ranked =
      [...candidates.entries()]
        .sort(
          (
            a,
            b
          ) => {

            if (
              b[1] !==
              a[1]
            ) {

              return (
                b[1] -
                a[1]
              );
            }


            return naturalCompare(
              a[0],
              b[0]
            );
          }
        );


    return ranked[0][0];
  }


  /* ============================================================
     BUILD ONE HUB REGISTRY ENTRY
  ============================================================ */

  /**
   * buildHubRegistryEntry()
   * Constructs a hub record containing its identity, planning URL, aircraft types and route catalogues.
   *
   * Notes: This describes the existing behaviour; no logic is changed.
   */
  function buildHubRegistryEntry(
    url,
    data
  ) {

    const hubId =
      Number(
        data?.hubAirportId
      );


    if (
      !Number.isFinite(
        hubId
      )
    ) {

      throw new Error(
        "Cannot build hub registry entry: invalid hubAirportId."
      );
    }


    const code =
      discoverHubCodeFromData(
        data
      );


    const typeCatalogue =
      buildAircraftTypeCatalogue(
        data
      );


    const routeCatalogues =
      buildRouteCatalogues(
        data
      );


    return {

      code,

      hubId,

      planningUrl:
        url,

      data,

      typeCatalogue,

      routeCatalogues,

      loadedAt:
        Date.now()
    };
  }


  /* ============================================================
     REGISTER ONE HUB

     If the same hub appears more than once in Performance
     resources, keep the most recently inspected working URL.
  ============================================================ */

  /**
   * registerHub()
   * Stores a discovered hub by airport code and logs its available aircraft, routes and types.
   *
   * Notes: This describes the existing behaviour; no logic is changed.
   */
  function registerHub(
    entry
  ) {

    if (!entry) {

      return;
    }


    const code =
      normaliseCode(
        entry.code
      );


    if (!code) {

      throw new Error(
        "Cannot register hub without an airport code."
      );
    }


    hubRegistry.set(
      code,
      {
        ...entry,

        code
      }
    );


    console.log(
      "[AM Rotation Builder] Registered hub:",
      {
        code:
          entry.code,

        hubId:
          entry.hubId,

        planningUrl:
          entry.planningUrl,

        aircraft:
          getAircraftArray(
            entry.data
          ).length,

        routes:
          entry.routeCatalogues
            ?.routes
            ?.size ??
          0,

        aircraftTypes:
          entry.typeCatalogue
            ?.size ??
          0
      }
    );
  }


  /* ============================================================
     DISCOVER ALL PLANNING DATASETS CURRENTLY AVAILABLE

     Returns the number of successfully registered hubs.
  ============================================================ */

  /**
   * discoverAvailableHubs()
   * Inspects discovered planning resources within a timeout and registers valid hubs.
   *
   * Notes: This describes the existing behaviour; no logic is changed.
   */
  async function discoverAvailableHubs(
    timeoutMs = 15000
  ) {

    const started =
      Date.now();


    const inspectedUrls =
      new Set();


    let lastNewResourceTime =
      Date.now();


    while (
      Date.now() -
      started <
      timeoutMs
    ) {

      const urls =
        getPlanningResourceUrls();


      let foundNewUrl =
        false;


      for (
        const url of
        urls
      ) {

        if (
          inspectedUrls.has(
            url
          )
        ) {

          continue;
        }


        inspectedUrls.add(
          url
        );


        foundNewUrl =
          true;


        lastNewResourceTime =
          Date.now();


        try {

          const data =
            await fetchPlanningUrl(
              url
            );


          const entry =
            buildHubRegistryEntry(
              url,
              data
            );


          registerHub(
            entry
          );


        } catch (error) {

          console.warn(
            "[AM Rotation Builder] Could not inspect planning dataset:",
            url,
            error
          );
        }
      }


      /*
       * Once we have at least one hub and no new planning
       * resource has appeared for a short period, discovery
       * can finish.
       */

      if (
        hubRegistry.size >
          0 &&
        !foundNewUrl &&
        Date.now() -
          lastNewResourceTime >
          1000
      ) {

        break;
      }


      await sleep(
        250
      );
    }


    return hubRegistry.size;
  }


  /* ============================================================
     HUB ACCESS
  ============================================================ */

  /**
   * getHubEntry()
   * Returns a previously discovered hub entry or null.
   *
   * Notes: This describes the existing behaviour; no logic is changed.
   */
  function getHubEntry(
    hubCode
  ) {

    const code =
      normaliseCode(
        hubCode
      );


    return (
      hubRegistry.get(
        code
      ) ||
      null
    );
  }


  /**
   * requireHubEntry()
   * Requires a hub to exist in the registry and reports available codes if not.
   *
   * Notes: This describes the existing behaviour; no logic is changed.
   */
  function requireHubEntry(
    hubCode
  ) {

    const code =
      normaliseCode(
        hubCode
      );


    const entry =
      getHubEntry(
        code
      );


    if (!entry) {

      const available =
        [...hubRegistry.keys()]
          .sort(
            naturalCompare
          );


      const suffix =
        available.length
          ? (
              ` Currently discovered hubs: ` +
              `${available.join(", ")}.`
            )
          : (
              " No planning hubs have been discovered."
            );


      throw new Error(
        `Hub ${code} is not currently available to the rotation builder.` +
        suffix
      );
    }


    return entry;
  }


  /* ============================================================
     FRESH HUB DATA

     Every preflight/write verification uses the exact planning
     URL previously locked to that hub.

     A response for a different hub ID is rejected.
  ============================================================ */

  /**
   * getFreshHubData()
   * Reloads one hub's planning JSON and verifies that both hub ID and airport code still match.
   *
   * Notes: This describes the existing behaviour; no logic is changed.
   */
  async function getFreshHubData(
    hubCode
  ) {

    const entry =
      requireHubEntry(
        hubCode
      );


    const freshData =
      await fetchPlanningUrl(
        entry.planningUrl,
        entry.hubId
      );


    const freshCode =
      discoverHubCodeFromData(
        freshData
      );


    if (
      normaliseCode(
        freshCode
      ) !==
      normaliseCode(
        entry.code
      )
    ) {

      throw new Error(
        `Fresh planning data expected ${entry.code}, ` +
        `but resolved as ${freshCode}.`
      );
    }


    return freshData;
  }


  /* ============================================================
     REFRESH ONE HUB REGISTRY ENTRY

     This is read-only.

     It rebuilds aircraft specifications and route catalogues
     from a fresh GET.
  ============================================================ */

  /**
   * refreshHubEntry()
   * Rebuilds a hub's cached aircraft and route catalogues from a fresh read-only request.
   *
   * Notes: This describes the existing behaviour; no logic is changed.
   */
  async function refreshHubEntry(
    hubCode
  ) {

    const oldEntry =
      requireHubEntry(
        hubCode
      );


    const freshData =
      await getFreshHubData(
        hubCode
      );


    const freshEntry = {

      code:
        oldEntry.code,

      hubId:
        oldEntry.hubId,

      planningUrl:
        oldEntry.planningUrl,

      data:
        freshData,

      typeCatalogue:
        buildAircraftTypeCatalogue(
          freshData
        ),

      routeCatalogues:
        buildRouteCatalogues(
          freshData
        ),

      loadedAt:
        Date.now()
    };


    hubRegistry.set(
      freshEntry.code,
      freshEntry
    );


    return freshEntry;
  }


  /* ============================================================
     REFRESH EVERY HUB REQUIRED BY A BATCH

     We deliberately refresh each hub once, not once per
     rotation.

     Example:

       BKK rotation 1
       BKK rotation 2
       BKK rotation 3
       BOM rotation 1

     requires only:

       one fresh BKK GET
       one fresh BOM GET
  ============================================================ */

  /**
   * refreshRequiredHubs()
   * Refreshes each distinct hub used in the batch once and records audit progress.
   *
   * Notes: This describes the existing behaviour; no logic is changed.
   */
  async function refreshRequiredHubs(
    requests,
    audit = null
  ) {

    const requiredHubs =
      getRequiredHubCodes(
        requests
      );


    const refreshed =
      new Map();


    for (
      const hub of
      requiredHubs
    ) {

      if (audit) {

        auditInfo(
          audit,
          `Loading fresh ${hub} planning data`
        );
      }


      const entry =
        await refreshHubEntry(
          hub
        );


      refreshed.set(
        hub,
        entry
      );


      if (audit) {

        auditOk(
          audit,
          `${hub}: fresh planning data loaded`
        );
      }
    }


    return refreshed;
  }


  /* ============================================================
     AIRCRAFT FILTERING BY HUB
  ============================================================ */

  /**
   * getHubAircraft()
   * Filters the planning aircraft collection to the requested hub's numeric ID.
   *
   * Notes: This describes the existing behaviour; no logic is changed.
   */
  function getHubAircraft(
    hubCode,
    data = null
  ) {

    const entry =
      requireHubEntry(
        hubCode
      );


    const sourceData =
      data ||
      entry.data;


    return getAircraftArray(
      sourceData
    )
      .filter(
        aircraft =>
          Number(
            aircraft.hubId
          ) ===
          Number(
            entry.hubId
          )
      );
  }


  /**
   * getAircraftTypesForHub()
   * Lists distinct aircraft models currently present at a hub.
   *
   * Notes: This describes the existing behaviour; no logic is changed.
   */
  function getAircraftTypesForHub(
    hubCode,
    data = null
  ) {

    const types =
      new Set();


    for (
      const aircraft of
      getHubAircraft(
        hubCode,
        data
      )
    ) {

      const type =
        String(
          aircraft?.aircraftListName ??
          ""
        ).trim();


      if (type) {

        types.add(
          type
        );
      }
    }


    return [
      ...types
    ].sort(
      naturalCompare
    );
  }


  /**
   * getEligibleAircraft()
   * Selects unused aircraft of a specified model at a hub and sorts them deterministically.
   *
   * Notes: This describes the existing behaviour; no logic is changed.
   */
  function getEligibleAircraft(
    hubCode,
    aircraftType,
    data = null
  ) {

    return getHubAircraft(
      hubCode,
      data
    )
      .filter(
        aircraft =>
          String(
            aircraft.aircraftListName
          ) ===
            String(
              aircraftType
            ) &&
          isAircraftEmpty(
            aircraft
          )
      )
      .sort(
        (
          a,
          b
        ) => {

          const nameCompare =
            naturalCompare(
              a.name ||
              "",
              b.name ||
              ""
            );


          if (
            nameCompare !==
            0
          ) {

            return nameCompare;
          }


          return (
            Number(
              a.id
            ) -
            Number(
              b.id
            )
          );
        }
      );
  }


  /* ============================================================
     AIRCRAFT POOL

     Multiple rotations in one request MUST NOT accidentally
     select the same seven aircraft.

     reservedAircraftIds is shared across the whole batch.

     Example:

       BKK A380 rotation 1 -> first 7 empty A380s
       BKK A380 rotation 2 -> next 7 empty A380s
       BKK A380 rotation 3 -> next 7 empty A380s

     This is deterministic.
  ============================================================ */

  /**
   * allocateSevenAircraft()
   * Reserves seven eligible aircraft for one rotation, preventing reuse across the batch.
   *
   * Notes: This describes the existing behaviour; no logic is changed.
   */
  function allocateSevenAircraft(
    hubCode,
    aircraftType,
    reservedAircraftIds,
    data = null
  ) {

    const eligible =
      getEligibleAircraft(
        hubCode,
        aircraftType,
        data
      );


    const available =
      eligible.filter(
        aircraft =>
          !reservedAircraftIds.has(
            aircraftReservationKey(
              aircraft.id
            )
          )
      );


    if (
      available.length <
      7
    ) {

      const alreadyReserved =
        eligible.length -
        available.length;


      throw new Error(
        `${hubCode}: cannot allocate another ${aircraftType} rotation. ` +
        `${eligible.length} empty aircraft are eligible, ` +
        `${alreadyReserved} are already reserved by earlier rotations ` +
        `in this batch, leaving ${available.length}. Seven are required.`
      );
    }


    const selected =
      available.slice(
        0,
        7
      );


    for (
      const aircraft of
      selected
    ) {

      reservedAircraftIds.add(
        aircraftReservationKey(
          aircraft.id
        )
      );
    }


    return {

      selected,

      eligibleCount:
        eligible.length,

      remainingAfterAllocation:
        available.length -
        7
    };
  }


  /* ============================================================
     VALIDATE REQUESTED AIRCRAFT TYPES

     Do this before allocating anything so a typo in a later
     rotation cannot leave us with a partially constructed
     batch.
  ============================================================ */

  /**
   * validateRequestedAircraftTypes()
   * Checks that every requested model exists in its hub's live aircraft catalogue.
   *
   * Notes: This describes the existing behaviour; no logic is changed.
   */
  function validateRequestedAircraftTypes(
    requests
  ) {

    for (
      const request of
      requests
    ) {

      const entry =
        requireHubEntry(
          request.hub
        );


      if (
        !entry.typeCatalogue.has(
          request.aircraftType
        )
      ) {

        const availableTypes =
          [
            ...entry.typeCatalogue.keys()
          ].sort(
            naturalCompare
          );


        throw new Error(
          `${request.rotationName}: aircraft type ` +
          `${request.aircraftType} was not found at ${request.hub}. ` +
          (
            availableTypes.length
              ? `Available types include: ${availableTypes.join(", ")}.`
              : "No aircraft types were found for this hub."
          )
        );
      }
    }


    return true;
  }


  /* ============================================================
     CAPACITY CHECK BEFORE GENERATION

     This produces a useful failure before individual
     allocations begin.

     Each rotation always consumes exactly seven empty aircraft.
  ============================================================ */

  /**
   * validateBatchAircraftCapacity()
   * Ensures each hub/model group has seven empty aircraft for every requested rotation.
   *
   * Notes: This describes the existing behaviour; no logic is changed.
   */
  function validateBatchAircraftCapacity(
    requests
  ) {

    const groups =
      new Map();


    for (
      const request of
      requests
    ) {

      const key =
        `${request.hub}|||${request.aircraftType}`;

      if (
        !groups.has(
          key
        )
      ) {

        groups.set(
          key,
          {
            hub:
              request.hub,

            aircraftType:
              request.aircraftType,

            rotations:
              0
          }
        );
      }


      groups.get(
        key
      ).rotations++;
    }


    for (
      const group of
      groups.values()
    ) {

      const eligible =
        getEligibleAircraft(
          group.hub,
          group.aircraftType
        );


      const required =
        group.rotations *
        7;


      if (
        eligible.length <
        required
      ) {

        throw new Error(
          `${group.hub}: batch requests ${group.rotations} ` +
          `${group.aircraftType} rotation` +
          `${group.rotations === 1 ? "" : "s"}, requiring ` +
          `${required} empty aircraft, but only ` +
          `${eligible.length} are currently eligible.`
        );
      }
    }


    return true;
  }


  /* ============================================================
     ROUTE RESOLUTION FOR ONE REQUEST
  ============================================================ */

  /**
   * resolveRequestedRoutes()
   * Resolves and range-validates every destination in one rotation request.
   *
   * Notes: This describes the existing behaviour; no logic is changed.
   */
  function resolveRequestedRoutes(
    request
  ) {

    const entry =
      requireHubEntry(
        request.hub
      );


    const spec =
      getAircraftTypeSpec(
        request.aircraftType,
        entry.typeCatalogue
      );


    const resolved =
      [];


    for (
      const destination of
      request.destinations
    ) {

      const route =
        resolveRoute(
          entry.routeCatalogues,
          request.aircraftType,
          request.hub,
          destination,
          entry.typeCatalogue
        );


      if (!route) {

        throw new Error(
            `${request.rotationName}: route ` +
            `${request.hub}/${destination} was not found ` +
            `in the live ${request.hub} planning data.`
        );
      }


      validateRange(
        route,
        spec
      );


      resolved.push(
        route
      );
    }


    return {

      aircraftSpec:
        spec,

      routes:
        resolved
    };
  }


  /* ============================================================
     BASE PATTERN

     Each destination represents:

       HUB -> DESTINATION -> HUB

     The next route starts immediately when the previous
     rotation finishes.

     The entire pattern must fit inside 168 hours.
  ============================================================ */

  /**
   * buildBasePattern()
   * Builds consecutive round trips starting Monday 00:00 and rejects patterns longer than 168 hours.
   *
   * Notes: This describes the existing behaviour; no logic is changed.
   */
  function buildBasePattern(
    hub,
    destinations,
    aircraftType,
    catalogues,
    typeCatalogue
  ) {

    const hubCode =
      normaliseCode(
        hub
      );


    const spec =
      getAircraftTypeSpec(
        aircraftType,
        typeCatalogue
      );


    const schedule =
      [];


    let cursor =
      0;


    for (
      const destination of
      destinations
    ) {

      const destinationCode =
        normaliseCode(
          destination
        );


      const route =
        resolveRoute(
          catalogues,
          aircraftType,
          hubCode,
          destinationCode,
          typeCatalogue
        );


if (!route) {

  throw new Error(
    `Could not find route ${hubCode}/${destinationCode}.`
  );
}

      validateRange(
        route,
        spec
      );


      const duration =
        Number(
          route.duration
        );


if (
  !Number.isFinite(
    duration
  ) ||
  duration <= 0
) {

  throw new Error(
    `${hubCode}/${destinationCode} has no valid duration.`
  );
}

      const takeOffTime =
        cursor;


      const landingTime =
        cursor +
        duration;


      schedule.push({

        index:
          schedule.length,

        route,

        lineId:
          Number(
            route.id
          ),

        takeOffTime,

        landingTime,

        duration
      });


      cursor =
        landingTime;
    }


    if (
      cursor >
      WEEK_SECONDS
    ) {

      throw new Error(
        `${hubCode} ${aircraftType} rotation requires ` +
        `${secondsToDuration(cursor)}, which exceeds the ` +
        `168h weekly limit by ` +
        `${secondsToDuration(cursor - WEEK_SECONDS)}.`
      );
    }


    return {

      hub:
        hubCode,

      aircraftType,

      schedule,

      totalDuration:
        cursor,

      unusedDuration:
        WEEK_SECONDS -
        cursor,

      utilisation:
        (
          cursor /
          WEEK_SECONDS
        ) *
        100
    };
  }


  /* ============================================================
     BUILD THE SEVEN DAY-SHIFTED AIRCRAFT

     Aircraft 01 -> Monday 00:00
     Aircraft 02 -> Tuesday 00:00
     ...
     Aircraft 07 -> Sunday 00:00

     Because the AM week is cyclic, all offsets are modulo
     604800.
  ============================================================ */

  /**
   * buildAircraftRotations()
   * Creates seven copies of a weekly pattern, offset by one day per assigned aircraft.
   *
   * Notes: This describes the existing behaviour; no logic is changed.
   */
  function buildAircraftRotations(
    selectedAircraft,
    basePattern,
    rotationName
  ) {

    if (
      !Array.isArray(
        selectedAircraft
      ) ||
      selectedAircraft.length !==
        7
    ) {

      throw new Error(
        "Exactly seven aircraft are required to build a rotation."
      );
    }


    return selectedAircraft.map(
      (
        plane,
        index
      ) => {

        const dayShift =
          index *
          DAY_SECONDS;


        const schedule =
          basePattern.schedule.map(
            item => ({

              route:
                item.route,

              lineId:
                Number(
                  item.lineId
                ),

              duration:
                Number(
                  item.duration
                ),

              takeOffTime:
                moduloWeek(
                  Number(
                    item.takeOffTime
                  ) +
                  dayShift
                ),

              landingTime:
                moduloWeek(
                  Number(
                    item.takeOffTime
                  ) +
                  dayShift +
                  Number(
                    item.duration
                  )
                )
            })
          );


        const newName =
          `${rotationName}-` +
          `${String(index + 1).padStart(2, "0")}`;


        const payload = {

          aircraftId:
            Number(
              plane.id
            ),

          added:
            schedule.map(
              item => ({

                takeOffTime:
                  Number(
                    item.takeOffTime
                  ),

                lineId:
                  Number(
                    item.lineId
                  )
              })
            )
        };


        return {

          rotationIndex:
            index,

          dayIndex:
            index,

          day:
            DAYS[index],

          aircraftId:
            Number(
              plane.id
            ),

          oldName:
            String(
              plane.name ??
              ""
            ),

          newName,

          aircraftType:
            basePattern.aircraftType,

          hub:
            basePattern.hub,

          schedule,

          payload
        };
      }
    );
  }


  /* ============================================================
     BUILD ONE COMPLETE ROTATION REQUEST
  ============================================================ */

  /**
   * buildRotationRequest()
   * Assembles one validated rotation request into the generated rotation structure.
   *
   * Notes: This describes the existing behaviour; no logic is changed.
   */
  function buildRotationRequest(
    request,
    reservedAircraftIds
  ) {

    const entry =
      requireHubEntry(
        request.hub
      );


    const {
      aircraftSpec,
      routes
    } =
      resolveRequestedRoutes(
        request
      );


    /*
     * Build from the resolved route catalogue again so the
     * exact same central validation path is used throughout
     * generation and later preflight.
     */

    const basePattern =
      buildBasePattern(
        request.hub,
        request.destinations,
        request.aircraftType,
        entry.routeCatalogues,
        entry.typeCatalogue
      );


    const allocation =
      allocateSevenAircraft(
        request.hub,
        request.aircraftType,
        reservedAircraftIds,
        entry.data
      );


    const rotations =
      buildAircraftRotations(
        allocation.selected,
        basePattern,
        request.rotationName
      );


    return {

      batchIndex:
        request.batchIndex,

      sourceLine:
        request.sourceLine,

      hub:
        request.hub,

      hubId:
        entry.hubId,

      planningUrl:
        entry.planningUrl,

      aircraftType:
        request.aircraftType,

      aircraftSpec:
        clone(
          aircraftSpec
        ),

      requiredConfig:
        request.requiredConfig
          ? clone(
              request.requiredConfig
            )
          : null,

      rotationName:
        request.rotationName,

      destinations:
        [
          ...request.destinations
        ],

      resolvedRoutes:
        routes.map(
          route => ({
            id:
              Number(
                route.id
              ),

            name:
              route.name,

            hub:
              route.hub,

            destination:
              route.destination,

            distance:
              Number(
                route.distance
              ),

            category:
              Number(
                route.category
              ),

            aircraftCategory:
              Number(
                route.aircraftCategory
              ),

            duration:
              Number(
                route.duration
              ),

            durationSource:
              route.durationSource
          })
        ),

      basePattern,

      eligibleCount:
        allocation.eligibleCount,

      remainingAfterAllocation:
        allocation.remainingAfterAllocation,

      lockedAircraft:
        allocation.selected.map(
          plane => ({

            id:
              Number(
                plane.id
              ),

            name:
              String(
                plane.name ??
                ""
              ),

            hubId:
              Number(
                plane.hubId
              ),

            aircraftListName:
              String(
                plane.aircraftListName ??
                ""
              )
          })
        ),

      rotations
    };
  }


  /* ============================================================
     BUILD WHOLE BATCH

     Aircraft IDs are reserved globally across every rotation
     in the request.

     This guarantees that one aircraft can never accidentally
     be assigned to two rotations in the same batch.
  ============================================================ */

  /**
   * buildRotationBatch()
   * Processes all requests into a batch while reserving aircraft across rotations.
   *
   * Notes: This describes the existing behaviour; no logic is changed.
   */
  function buildRotationBatch(
    requests
  ) {

    validateRequestedAircraftTypes(
      requests
    );


    validateBatchAircraftCapacity(
      requests
    );


    const reservedAircraftIds =
      new Set();


    const rotations =
      [];


    for (
      const request of
      requests
    ) {

      const generated =
        buildRotationRequest(
          request,
          reservedAircraftIds
        );


      rotations.push(
        generated
      );
    }


    const totalAircraft =
      rotations.reduce(
        (
          total,
          rotation
        ) =>
          total +
          rotation.rotations.length,
        0
      );


    const totalFlights =
      rotations.reduce(
        (
          total,
          rotation
        ) =>
          total +
          rotation.rotations.reduce(
            (
              aircraftTotal,
              aircraftRotation
            ) =>
              aircraftTotal +
              aircraftRotation.schedule.length,
            0
          ),
        0
      );


    return {

      version:
        VERSION,

      generatedAt:
        Date.now(),

      requestFingerprint:
        makeBatchRequestFingerprint(
          requests
        ),

      requests:
        clone(
          requests
        ),

      hubs:
        getRequiredHubCodes(
          requests
        ),

      rotations,

      totalRotations:
        rotations.length,

      totalAircraft,

      totalFlights,

      reservedAircraftIds:
        [
          ...reservedAircraftIds
        ]
    };
  }


  /* ============================================================
     END CHUNK 2
  ============================================================ */

   /* ============================================================
     CYCLIC WEEKLY SCHEDULE VALIDATION

     Airlines Manager schedules repeat every 604800 seconds.

     A flight can therefore cross the Sunday -> Monday boundary.

     Example:

       Sunday 22:00
       duration 4h

     occupies:

       Sunday 22:00 -> Monday 00:00
       Monday 00:00 -> Monday 02:00

     To test overlaps correctly we split any wrapping interval
     into ordinary intervals inside the weekly range.
  ============================================================ */

  /**
   * splitCyclicInterval()
   * Splits a weekly interval that crosses the week boundary into comparable segments.
   *
   * Notes: This describes the existing behaviour; no logic is changed.
   */
  function splitCyclicInterval(
    start,
    duration
  ) {

    const normalisedStart =
      moduloWeek(
        Number(
          start
        )
      );


    const numericDuration =
      Number(
        duration
      );


    if (
      !Number.isFinite(
        normalisedStart
      ) ||
      !Number.isFinite(
        numericDuration
      ) ||
      numericDuration <= 0
    ) {

      throw new Error(
        "Cannot validate an invalid schedule interval."
      );
    }


    if (
      numericDuration >
      WEEK_SECONDS
    ) {

      throw new Error(
        "A single flight rotation cannot exceed 168 hours."
      );
    }


    const end =
      normalisedStart +
      numericDuration;


    if (
      end <=
      WEEK_SECONDS
    ) {

      return [
        {
          start:
            normalisedStart,

          end
        }
      ];
    }


    return [
      {
        start:
          normalisedStart,

        end:
          WEEK_SECONDS
      },

      {
        start:
          0,

        end:
          end -
          WEEK_SECONDS
      }
    ];
  }


  /**
   * intervalsOverlap()
   * Tests whether two time intervals intersect.
   *
   * Notes: This describes the existing behaviour; no logic is changed.
   */
  function intervalsOverlap(
    a,
    b
  ) {

    return (
      Number(
        a.start
      ) <
        Number(
          b.end
        ) &&
      Number(
        b.start
      ) <
        Number(
          a.end
        )
    );
  }


  /**
   * schedulesOverlap()
   * Checks whether two flight schedules contain overlapping occupied time.
   *
   * Notes: This describes the existing behaviour; no logic is changed.
   */
  function schedulesOverlap(
    first,
    second
  ) {

    const firstIntervals =
      splitCyclicInterval(
        first.takeOffTime,
        first.duration
      );


    const secondIntervals =
      splitCyclicInterval(
        second.takeOffTime,
        second.duration
      );


    for (
      const a of
      firstIntervals
    ) {

      for (
        const b of
        secondIntervals
      ) {

        if (
          intervalsOverlap(
            a,
            b
          )
        ) {

          return true;
        }
      }
    }


    return false;
  }


  /* ============================================================
     VALIDATE ONE AIRCRAFT'S WEEK

     The same aircraft must never have two overlapping
     rotations.

     Back-to-back flights are allowed:

       flight A ends 12:00
       flight B starts 12:00

     because intervals are treated as:

       [start, end)
  ============================================================ */

  /**
   * validateRotationSchedule()
   * Checks a generated rotation's timing and scheduling invariants.
   *
   * Notes: This describes the existing behaviour; no logic is changed.
   */
  function validateRotationSchedule(
    aircraftRotation
  ) {

    if (!aircraftRotation) {

      throw new Error(
        "Cannot validate an empty aircraft rotation."
      );
    }


    const schedule =
      aircraftRotation.schedule;


if (
  !Array.isArray(
    schedule
  ) ||
  schedule.length === 0
) {

  throw new Error(
    `Aircraft ID ${aircraftRotation.aircraftId} ` +
    `has no scheduled flights.`
  );
}

    for (
      let i = 0;
      i < schedule.length;
      i++
    ) {

      const item =
        schedule[i];


      const takeOffTime =
        Number(
          item.takeOffTime
        );


      const duration =
        Number(
          item.duration
        );


      const lineId =
        Number(
          item.lineId
        );


      if (
        !Number.isFinite(
          takeOffTime
        ) ||
        takeOffTime < 0 ||
        takeOffTime >=
          WEEK_SECONDS
      ) {

        throw new Error(
          `${aircraftRotation.newName}: route ${i + 1} ` +
          `has invalid take-off time ${item.takeOffTime}.`
        );
      }


      if (
        !Number.isSafeInteger(
  duration
) ||
        duration <= 0 ||
        duration >
          WEEK_SECONDS
      ) {

        throw new Error(
          `${aircraftRotation.newName}: route ${i + 1} ` +
          `has invalid duration ${item.duration}.`
        );
      }


      if (
        !Number.isSafeInteger(
  lineId
        ) ||
        lineId <= 0
      ) {

        throw new Error(
          `${aircraftRotation.newName}: route ${i + 1} ` +
          `has invalid line ID ${item.lineId}.`
        );
      }
    }


    for (
      let i = 0;
      i < schedule.length;
      i++
    ) {

      for (
        let j = i + 1;
        j < schedule.length;
        j++
      ) {

        if (
          schedulesOverlap(
            schedule[i],
            schedule[j]
          )
        ) {

          const firstName =
            schedule[i]
              ?.route
              ?.name ||
            `line ${schedule[i].lineId}`;


          const secondName =
            schedule[j]
              ?.route
              ?.name ||
            `line ${schedule[j].lineId}`;


          throw new Error(
            `${aircraftRotation.newName} has an overlapping ` +
            `weekly schedule: ${firstName} and ${secondName}.`
          );
        }
      }
    }


    return true;
  }


  /* ============================================================
     VALIDATE ALL SEVEN AIRCRAFT IN ONE ROTATION
  ============================================================ */

  /**
   * validateAllRotationSchedules()
   * Applies schedule validation to every rotation in the generated batch.
   *
   * Notes: This describes the existing behaviour; no logic is changed.
   */
  function validateAllRotationSchedules(
    rotations
  ) {

    if (
      !Array.isArray(
        rotations
      ) ||
      rotations.length !==
        7
    ) {

      throw new Error(
        "A rotation must contain exactly seven aircraft schedules."
      );
    }


    for (
      const rotation of
      rotations
    ) {

      validateRotationSchedule(
        rotation
      );
    }


    return true;
  }


  /* ============================================================
     CANONICAL SCHEDULE

     Airlines Manager does not need to return planningList in
     the same array order as our POST.

     Therefore comparisons use a stable canonical order:

       takeOffTime
       lineId

     Only those two fields define the desired native schedule.
  ============================================================ */

  /**
   * canonicalSchedule()
   * Converts a schedule to a stable representation suitable for comparison.
   *
   * Notes: This describes the existing behaviour; no logic is changed.
   */
  function canonicalSchedule(
    schedule
  ) {

    if (
      !Array.isArray(
        schedule
      )
    ) {

      return [];
    }


    return schedule
      .map(
        item => ({

          takeOffTime:
            Number(
              item.takeOffTime
            ),

          lineId:
            Number(
              item.lineId
            )
        })
      )
      .sort(
        (
          a,
          b
        ) => {

          if (
            a.takeOffTime !==
            b.takeOffTime
          ) {

            return (
              a.takeOffTime -
              b.takeOffTime
            );
          }


          return (
            a.lineId -
            b.lineId
          );
        }
      );
  }


  /**
   * schedulesEqual()
   * Compares two schedules using their canonical forms.
   *
   * Notes: This describes the existing behaviour; no logic is changed.
   */
  function schedulesEqual(
    first,
    second
  ) {

    return (
      JSON.stringify(
        canonicalSchedule(
          first
        )
      ) ===
      JSON.stringify(
        canonicalSchedule(
          second
        )
      )
    );
  }


  /* ============================================================
     CANONICAL NATIVE AM PAYLOAD
  ============================================================ */

  /**
   * canonicalPayload()
   * Normalises a scheduling request payload for reliable comparison.
   *
   * Notes: This describes the existing behaviour; no logic is changed.
   */
  function canonicalPayload(
    payload
  ) {

    return {

      aircraftId:
        Number(
          payload?.aircraftId
        ),

      added:
        canonicalSchedule(
          payload?.added
        )
    };
  }


  /* ============================================================
     AIRCRAFT ID UNIQUENESS

     This is especially important in v1.0 because one batch can
     contain many rotations and many hubs.

     No permanent aircraft ID may occur more than once anywhere
     in the batch.
  ============================================================ */

  /**
   * getBatchAircraftIds()
   * Collects the permanent IDs of all aircraft assigned across the batch.
   *
   * Notes: This describes the existing behaviour; no logic is changed.
   */
  function getBatchAircraftIds(
    batch
  ) {

    const ids =
      [];


    for (
      const generatedRotation of
      batch?.rotations ||
      []
    ) {

      for (
        const aircraftRotation of
        generatedRotation.rotations ||
        []
      ) {

        ids.push(
          Number(
            aircraftRotation.aircraftId
          )
        );
      }
    }


    return ids;
  }


  /**
   * validateUniqueBatchAircraft()
   * Rejects any aircraft ID assigned more than once across the batch.
   *
   * Notes: This describes the existing behaviour; no logic is changed.
   */
  function validateUniqueBatchAircraft(
    batch
  ) {

    const ids =
      getBatchAircraftIds(
        batch
      );


    const seen =
      new Set();


    for (
      const id of
      ids
    ) {

      if (
        !Number.isFinite(
          id
        ) ||
        id <= 0
      ) {

        throw new Error(
          `Batch contains an invalid permanent aircraft ID: ${id}.`
        );
      }


      const key =
        aircraftReservationKey(
          id
        );


      if (
        seen.has(
          key
        )
      ) {

        throw new Error(
          `Permanent aircraft ID ${id} has been allocated ` +
          `more than once in this batch.`
        );
      }


      seen.add(
        key
      );
    }


    return true;
  }


  /* ============================================================
     ROTATION NAME UNIQUENESS

     We do not allow two requested rotations to generate the
     same seven names.

     For example, these cannot coexist:

       BKK-C1001
       BKK-C1001

     because both would try to create:

       BKK-C1001-01 ... BKK-C1001-07
  ============================================================ */

  /**
   * validateUniqueRotationNames()
   * Ensures rotation names remain unique in generated output.
   *
   * Notes: This describes the existing behaviour; no logic is changed.
   */
  function validateUniqueRotationNames(
    batch
  ) {

    const seen =
      new Set();


    for (
      const rotation of
      batch?.rotations ||
      []
    ) {

      const name =
        String(
          rotation.rotationName ??
          ""
        )
          .trim()
          .toUpperCase();


      if (!name) {

        throw new Error(
          "Batch contains a rotation without a name."
        );
      }


      if (
        seen.has(
          name
        )
      ) {

        throw new Error(
          `Rotation name ${rotation.rotationName} occurs ` +
          `more than once in this batch.`
        );
      }


      seen.add(
        name
      );
    }


    return true;
  }


  /* ============================================================
     GENERATED AIRCRAFT NAME UNIQUENESS

     This is a second independent check.

     Even if the base rotation names differ unexpectedly,
     no two generated aircraft may receive the same final name.
  ============================================================ */

  /**
   * validateUniqueGeneratedAircraftNames()
   * Rejects duplicate proposed aircraft names in generated rotations.
   *
   * Notes: This describes the existing behaviour; no logic is changed.
   */
  function validateUniqueGeneratedAircraftNames(
    batch
  ) {

    const seen =
      new Map();


    for (
      const generatedRotation of
      batch?.rotations ||
      []
    ) {

      for (
        const aircraftRotation of
        generatedRotation.rotations ||
        []
      ) {

        const name =
          String(
            aircraftRotation.newName ??
            ""
          )
            .trim()
            .toUpperCase();


        if (!name) {

          throw new Error(
            `Aircraft ID ${aircraftRotation.aircraftId} ` +
            `has no generated aircraft name.`
          );
        }


        if (
          seen.has(
            name
          )
        ) {

          const otherId =
            seen.get(
              name
            );


          throw new Error(
            `Generated aircraft name ${aircraftRotation.newName} ` +
            `would be assigned to both aircraft ID ${otherId} ` +
            `and aircraft ID ${aircraftRotation.aircraftId}.`
          );
        }


        seen.set(
          name,
          aircraftRotation.aircraftId
        );
      }
    }


    return true;
  }


  /* ============================================================
     VALIDATE ONE GENERATED ROTATION STRUCTURE
  ============================================================ */

  /**
   * validateGeneratedRotationStructure()
   * Checks the shape and required fields of generated rotation objects.
   *
   * Notes: This describes the existing behaviour; no logic is changed.
   */
  function validateGeneratedRotationStructure(
    generated
  ) {

    if (!generated) {

      throw new Error(
        "Generated rotation is missing."
      );
    }


    const hub =
      normaliseCode(
        generated.hub
      );


    if (!hub) {

      throw new Error(
        "Generated rotation has no hub."
      );
    }


    if (
      !generated.aircraftType
    ) {

      throw new Error(
        `${generated.rotationName || hub}: ` +
        `generated rotation has no aircraft type.`
      );
    }


    if (
      !generated.rotationName
    ) {

      throw new Error(
        `${hub}: generated rotation has no rotation name.`
      );
    }


    if (
      !Array.isArray(
        generated.destinations
      ) ||
      generated.destinations.length ===
        0
    ) {

      throw new Error(
        `${generated.rotationName}: no destinations were generated.`
      );
    }


    if (
      !generated.basePattern ||
      !Array.isArray(
        generated.basePattern.schedule
      )
    ) {

      throw new Error(
        `${generated.rotationName}: base pattern is missing.`
      );
    }


const baseDuration =
  Number(
    generated.basePattern.totalDuration
  );


if (
  !Number.isFinite(
    baseDuration
  ) ||
  baseDuration <= 0 ||
  baseDuration > WEEK_SECONDS
) {

  throw new Error(
    `${generated.rotationName}: base pattern has an invalid ` +
    `total duration or exceeds 168 hours.`
  );
}

    if (
      !Array.isArray(
        generated.lockedAircraft
      ) ||
      generated.lockedAircraft.length !==
        7
    ) {

      throw new Error(
        `${generated.rotationName}: exactly seven locked aircraft ` +
        `are required.`
      );
    }


    if (
      !Array.isArray(
        generated.rotations
      ) ||
      generated.rotations.length !==
        7
    ) {

      throw new Error(
        `${generated.rotationName}: exactly seven generated ` +
        `aircraft schedules are required.`
      );
    }


    const lockedIds =
      generated.lockedAircraft.map(
        aircraft =>
          Number(
            aircraft.id
          )
      );


    const rotationIds =
      generated.rotations.map(
        aircraft =>
          Number(
            aircraft.aircraftId
          )
      );


    if (
      JSON.stringify(
        lockedIds
      ) !==
      JSON.stringify(
        rotationIds
      )
    ) {

      throw new Error(
        `${generated.rotationName}: locked aircraft IDs do not ` +
        `match generated schedule aircraft IDs.`
      );
    }


    validateAllRotationSchedules(
      generated.rotations
    );


    return true;
  }


  /* ============================================================
     NATIVE PAYLOAD STRUCTURE VALIDATION

     Every aircraft write must contain:

       {
         aircraftId,
         added: [
           { takeOffTime, lineId },
           ...
         ]
       }

     Nothing else is required by the native AM schedule POST.
  ============================================================ */

  /**
   * validateNativePayload()
   * Validates one game-native scheduling payload before it can be submitted.
   *
   * Notes: This describes the existing behaviour; no logic is changed.
   */
  function validateNativePayload(
    payload,
    expectedAircraftId = null
  ) {

    if (
      !payload ||
      typeof payload !==
        "object"
    ) {

      throw new Error(
        "Native schedule payload is missing."
      );
    }

        /*
 * v1.0 — Schedule-only payload protection.
 *
 * Reject unexpected top-level fields to prevent
 * aircraft configuration data being submitted.
 */

const allowedFields =
  new Set([
    "aircraftId",
    "added"
  ]);


for (
  const field of Object.keys(payload)
) {

  if (
    !allowedFields.has(field)
  ) {

    throw new Error(
      `Native schedule payload contains unexpected field "${field}".`
    );
  }
}


const aircraftId =
  Number(
    payload.aircraftId
  );


if (
  !Number.isSafeInteger(
    aircraftId
  ) ||
  aircraftId <= 0
) {

  throw new Error(
    "Native schedule payload has an invalid aircraftId."
  );
}
    if (
      expectedAircraftId !==
        null &&
      aircraftId !==
        Number(
          expectedAircraftId
        )
    ) {

      throw new Error(
        `Native payload aircraft ID ${aircraftId} does not match ` +
        `expected aircraft ID ${expectedAircraftId}.`
      );
    }


if (
  !Array.isArray(
    payload.added
  ) ||
  payload.added.length === 0
) {

  throw new Error(
    `Aircraft ID ${aircraftId}: native payload has no scheduled flights.`
  );
}

    for (
      const item of
      payload.added
    ) {

        if (
  !item ||
  typeof item !== "object" ||
  Array.isArray(item)
) {

  throw new Error(
    `Aircraft ID ${aircraftId}: invalid schedule entry.`
  );
}

const allowedScheduleFields =
  new Set([
    "takeOffTime",
    "lineId"
  ]);

for (
  const field of Object.keys(item)
) {

  if (
    !allowedScheduleFields.has(field)
  ) {

    throw new Error(
      `Aircraft ID ${aircraftId}: unexpected schedule field "${field}".`
    );
  }
}

      const takeOffTime =
        Number(
          item?.takeOffTime
        );


      const lineId =
        Number(
          item?.lineId
        );


      if (
        !Number.isSafeInteger(
  takeOffTime
) ||
        takeOffTime < 0 ||
        takeOffTime >=
          WEEK_SECONDS
      ) {

        throw new Error(
          `Aircraft ID ${aircraftId}: invalid payload takeOffTime ` +
          `${item?.takeOffTime}.`
        );
      }


if (
  !Number.isSafeInteger(
    lineId
  ) ||
  lineId <= 0
) {

  throw new Error(
    `Aircraft ID ${aircraftId}: invalid payload lineId ` +
    `${item?.lineId}.`
  );
}
    }


    return true;
  }


  /* ============================================================
     VALIDATE PAYLOADS AGAINST GENERATED SCHEDULES
  ============================================================ */

  /**
   * validateRotationPayloads()
   * Checks every scheduling payload associated with a rotation.
   *
   * Notes: This describes the existing behaviour; no logic is changed.
   */
  function validateRotationPayloads(
    generated
  ) {

    for (
      const aircraftRotation of
      generated.rotations
    ) {

      validateNativePayload(
        aircraftRotation.payload,
        aircraftRotation.aircraftId
      );


      const expected =
        aircraftRotation.schedule.map(
          item => ({

            takeOffTime:
              Number(
                item.takeOffTime
              ),

            lineId:
              Number(
                item.lineId
              )
          })
        );


      if (
        !schedulesEqual(
          aircraftRotation.payload.added,
          expected
        )
      ) {

        throw new Error(
          `${generated.rotationName}: native payload for aircraft ` +
          `ID ${aircraftRotation.aircraftId} does not exactly match ` +
          `its generated schedule.`
        );
      }
    }


    return true;
  }


  /* ============================================================
     WHOLE-BATCH STRUCTURAL VALIDATION

     This is still completely read-only.

     It validates what we have generated before any live
     preflight is even possible.
  ============================================================ */

  /**
   * validateGeneratedBatch()
   * Runs structural and consistency checks on the entire generated batch.
   *
   * Notes: This describes the existing behaviour; no logic is changed.
   */
  function validateGeneratedBatch(
    batch
  ) {

    if (
      !batch ||
      !Array.isArray(
        batch.rotations
      ) ||
      batch.rotations.length ===
        0
    ) {

      throw new Error(
        "No generated rotations are available."
      );
    }


    if (
      Number(
        batch.totalRotations
      ) !==
      batch.rotations.length
    ) {

      throw new Error(
        "Generated batch rotation count is inconsistent."
      );
    }


    validateUniqueRotationNames(
      batch
    );


    validateUniqueBatchAircraft(
      batch
    );


    validateUniqueGeneratedAircraftNames(
      batch
    );


    let countedAircraft =
      0;


    let countedFlights =
      0;


    for (
      const generated of
      batch.rotations
    ) {

      validateGeneratedRotationStructure(
        generated
      );


      validateRotationPayloads(
        generated
      );


      countedAircraft +=
        generated.rotations.length;


      countedFlights +=
        generated.rotations.reduce(
          (
            total,
            aircraftRotation
          ) =>
            total +
            aircraftRotation.schedule.length,
          0
        );
    }


    if (
      countedAircraft !==
      Number(
        batch.totalAircraft
      )
    ) {

      throw new Error(
        `Generated batch reports ${batch.totalAircraft} aircraft, ` +
        `but ${countedAircraft} were actually validated.`
      );
    }


    if (
      countedFlights !==
      Number(
        batch.totalFlights
      )
    ) {

      throw new Error(
        `Generated batch reports ${batch.totalFlights} weekly flights, ` +
        `but ${countedFlights} were actually validated.`
      );
    }


    if (
      countedAircraft !==
      batch.rotations.length *
      7
    ) {

      throw new Error(
        "Generated batch does not contain exactly seven aircraft " +
        "per rotation."
      );
    }


    return {

      totalRotations:
        batch.rotations.length,

      totalAircraft:
        countedAircraft,

      totalFlights:
        countedFlights
    };
  }


  /* ============================================================
     STABLE FINGERPRINT SERIALISATION

     We fingerprint the safety-critical parts of the generated
     batch.

     The fingerprint is not intended to be cryptographic.

     Its purpose is to ensure:

       Generate
          ↓
       Live Preflight
          ↓
       Deploy

     all refer to the exact same generated batch.
  ============================================================ */

  /**
   * makeRotationFingerprintData()
   * Extracts the rotation properties used to detect changes between generation and deployment.
   *
   * Notes: This describes the existing behaviour; no logic is changed.
   */
  function makeRotationFingerprintData(
    generated
  ) {

    return {

      hub:
        normaliseCode(
          generated.hub
        ),

      hubId:
        Number(
          generated.hubId
        ),

      aircraftType:
        String(
          generated.aircraftType
        ),

      requiredConfig:
        generated.requiredConfig
          ? {
              eco:
                Number(
                  generated.requiredConfig.eco
                ),

              bus:
                Number(
                  generated.requiredConfig.bus
                ),

              first:
                Number(
                  generated.requiredConfig.first
                ),

              cargo:
                Number(
                  generated.requiredConfig.cargo
                )
            }
          : null,

      rotationName:
        String(
          generated.rotationName
        ),

      destinations:
        generated.destinations.map(
          normaliseCode
        ),

      totalDuration:
        Number(
          generated.basePattern
            ?.totalDuration
        ),

      routes:
        generated.resolvedRoutes.map(
          route => ({

            id:
              Number(
                route.id
              ),

            destination:
              normaliseCode(
                route.destination
              ),

            distance:
              Number(
                route.distance
              ),

            category:
              Number(
                route.category
              ),

            duration:
              Number(
                route.duration
              )
          })
        ),

      aircraft:
        generated.rotations.map(
          aircraftRotation => ({

            aircraftId:
              Number(
                aircraftRotation.aircraftId
              ),

            oldName:
              String(
                aircraftRotation.oldName
              ),

            newName:
              String(
                aircraftRotation.newName
              ),

            payload:
              canonicalPayload(
                aircraftRotation.payload
              )
          })
        )
    };
  }


  /**
   * makeRotationFingerprint()
   * Creates a stable fingerprint for a single generated rotation.
   *
   * Notes: This describes the existing behaviour; no logic is changed.
   */
  function makeRotationFingerprint(
    generated
  ) {

    return JSON.stringify(
      makeRotationFingerprintData(
        generated
      )
    );
  }


  /**
   * makeBatchFingerprintData()
   * Collects the batch properties that must remain unchanged during preflight and deployment.
   *
   * Notes: This describes the existing behaviour; no logic is changed.
   */
  function makeBatchFingerprintData(
    batch
  ) {

    return {

      version:
        String(
          batch.version
        ),

      requestFingerprint:
        String(
          batch.requestFingerprint
        ),

      hubs:
        [
          ...batch.hubs
        ]
          .map(
            normaliseCode
          )
          .sort(
            naturalCompare
          ),

      totalRotations:
        Number(
          batch.totalRotations
        ),

      totalAircraft:
        Number(
          batch.totalAircraft
        ),

      totalFlights:
        Number(
          batch.totalFlights
        ),

      rotations:
        batch.rotations.map(
          generated =>
            makeRotationFingerprintData(
              generated
            )
        )
    };
  }


  /**
   * makeBatchFingerprint()
   * Creates a stable fingerprint of the generated batch.
   *
   * Notes: This describes the existing behaviour; no logic is changed.
   */
  function makeBatchFingerprint(
    batch
  ) {

    return JSON.stringify(
      makeBatchFingerprintData(
        batch
      )
    );
  }


  /* ============================================================
     ASSERT BATCH HAS NOT CHANGED

     Used after asynchronous validation steps.
  ============================================================ */

  /**
   * assertBatchFingerprint()
   * Fails if the current batch no longer matches the expected fingerprint.
   *
   * Notes: This describes the existing behaviour; no logic is changed.
   */
  function assertBatchFingerprint(
    batch,
    expectedFingerprint
  ) {

    const current =
      makeBatchFingerprint(
        batch
      );


    if (
      current !==
      expectedFingerprint
    ) {

      throw new Error(
        "The generated batch changed during validation. " +
        "Generate the batch again."
      );
    }


    return true;
  }


  /* ============================================================

  /**

  function getAircraftFromPlanningData(

    return (

              aircraft.id
            ) ===

    );
  }


  /* ============================================================

     Convert one generated aircraft schedule to the exact
     structure used for schedule comparison.
  ============================================================ */

  /**
   * expectedPlanningEntries()
   * Builds the expected planning entries for comparison with live data.
   *
   * Notes: This describes the existing behaviour; no logic is changed.
   */
  function expectedPlanningEntries(
    aircraftRotation
  ) {

    return aircraftRotation.schedule.map(
      item => ({

        takeOffTime:
          Number(
            item.takeOffTime
          ),

        lineId:
          Number(
            item.lineId
          )
      })
    );
  }


  /* ============================================================
     VERIFY ONE AIRCRAFT'S LIVE SCHEDULE

     Returns a useful result object rather than throwing so the
     deployment code can distinguish:

       exact
       mismatch
       aircraft missing
  ============================================================ */

  /**
   * verifyAircraftScheduleInData()
   * Compares an aircraft's schedule in planning data against the expected schedule.
   *
   * Notes: This describes the existing behaviour; no logic is changed.
   */
  function verifyAircraftScheduleInData(
    data,
    aircraftRotation
  ) {

    const aircraft =
      getAircraftFromPlanningData(
        data,
        aircraftRotation.aircraftId
      );


    if (!aircraft) {

      return {

        ok:
          false,

        reason:
          "aircraft-missing",

        aircraft:
          null,

        expected:
          canonicalSchedule(
            expectedPlanningEntries(
              aircraftRotation
            )
          ),

        actual:
          []
      };
    }


    const expected =
      canonicalSchedule(
        expectedPlanningEntries(
          aircraftRotation
        )
      );


    const actual =
      canonicalSchedule(
        aircraft.planningList ||
        []
      );


    return {

      ok:
        JSON.stringify(
          expected
        ) ===
        JSON.stringify(
          actual
        ),

      reason:
        JSON.stringify(
          expected
        ) ===
        JSON.stringify(
          actual
        )
          ? "exact"
          : "schedule-mismatch",

      aircraft,

      expected,

      actual
    };
  }


  /* ============================================================
     VERIFY LOCKED AIRCRAFT AGAINST ONE FRESH HUB DATASET

     This is the multi-hub replacement for the old BKK-only
     verifyFreshAircraft path.

     Every permanent ID must:

       still exist
       still belong to the same hub
       still be the requested aircraft type
       still be empty
       still have its original name

     No substitutions are permitted.
  ============================================================ */

  /**
   * verifyFreshRotationAircraft()
   * Rechecks the assigned aircraft for a rotation against fresh live planning data.
   *
   * Notes: This describes the existing behaviour; no logic is changed.
   */
  function verifyFreshRotationAircraft(
    generated,
    freshData,
    audit = null
  ) {

    const expectedHubId =
      Number(
        generated.hubId
      );


    const locked =
      generated.lockedAircraft;


    if (
      !Array.isArray(
        locked
      ) ||
      locked.length !==
        7
    ) {

      throw new Error(
        `${generated.rotationName}: seven locked aircraft are required.`
      );
    }


    const liveAircraft =
      [];


    for (
      const lockedAircraft of
      locked
    ) {

      const aircraft =
        getAircraftFromPlanningData(
          freshData,
          lockedAircraft.id
        );


      if (!aircraft) {

        throw new Error(
          `${generated.rotationName}: permanent aircraft ID ` +
          `${lockedAircraft.id} is no longer present in fresh ` +
          `${generated.hub} planning data.`
        );
      }


      liveAircraft.push(
        aircraft
      );
    }


    if (audit) {

      auditOk(
        audit,
        `${generated.rotationName}: 7/7 locked permanent aircraft IDs found`
      );
    }


    const ids =
      liveAircraft.map(
        aircraft =>
          Number(
            aircraft.id
          )
      );


    if (
      new Set(
        ids
      ).size !==
      7
    ) {

      throw new Error(
        `${generated.rotationName}: locked permanent aircraft IDs ` +
        `are not unique.`
      );
    }


    if (audit) {

      auditOk(
        audit,
        `${generated.rotationName}: 7/7 permanent aircraft IDs are unique`
      );
    }


    for (
      const aircraft of
      liveAircraft
    ) {

      if (
        Number(
          aircraft.hubId
        ) !==
        expectedHubId
      ) {

        throw new Error(
          `${generated.rotationName}: aircraft ID ${aircraft.id} ` +
          `is no longer at ${generated.hub}.`
        );
      }
    }


    if (audit) {

      auditOk(
        audit,
        `${generated.rotationName}: 7/7 aircraft still at ${generated.hub}`
      );
    }


    for (
      const aircraft of
      liveAircraft
    ) {

      if (
        String(
          aircraft.aircraftListName
        ) !==
        String(
          generated.aircraftType
        )
      ) {

        throw new Error(
          `${generated.rotationName}: aircraft ID ${aircraft.id} ` +
          `is now ${aircraft.aircraftListName}, not ` +
          `${generated.aircraftType}.`
        );
      }
    }


    if (audit) {

      auditOk(
        audit,
        `${generated.rotationName}: 7/7 aircraft are still ` +
        `${generated.aircraftType}`
      );
    }


    for (
      const aircraft of
      liveAircraft
    ) {

      if (
        !isAircraftEmpty(
          aircraft
        )
      ) {

        throw new Error(
          `${generated.rotationName}: aircraft ID ${aircraft.id} ` +
          `is no longer empty.`
        );
      }
    }


    if (audit) {

      auditOk(
        audit,
        `${generated.rotationName}: 7/7 aircraft are still empty`
      );
    }


    for (
      let i = 0;
      i < liveAircraft.length;
      i++
    ) {

      const actualName =
        String(
          liveAircraft[i].name ??
          ""
        );


      const expectedName =
        String(
          locked[i].name ??
          ""
        );


      if (
        actualName !==
        expectedName
      ) {

        throw new Error(
          `${generated.rotationName}: aircraft ID ` +
          `${liveAircraft[i].id} was renamed from ` +
          `${expectedName} to ${actualName} after generation.`
        );
      }
    }


    if (audit) {

      auditOk(
        audit,
        `${generated.rotationName}: 7/7 original aircraft names unchanged`
      );
    }


    return liveAircraft;
  }


  /* ============================================================
     VERIFY THERE ARE NO AIRCRAFT COLLISIONS IN FRESH DATA

     The generated batch already prevents duplicate IDs.

     This additional check verifies the fresh locked-aircraft
     collections from every hub before deployment.
  ============================================================ */

  /**
   * validateFreshAircraftCollections()
   * Checks aircraft collections obtained during fresh validation.
   *
   * Notes: This describes the existing behaviour; no logic is changed.
   */
  function validateFreshAircraftCollections(
    freshRotationResults
  ) {

    const seen =
      new Set();


    for (
      const result of
      freshRotationResults
    ) {

      for (
        const aircraft of
        result.liveLockedAircraft
      ) {

        const key =
          aircraftReservationKey(
            aircraft.id
          );


        if (
          seen.has(
            key
          )
        ) {

          throw new Error(
            `Fresh validation found permanent aircraft ID ` +
            `${aircraft.id} in more than one requested rotation.`
          );
        }


        seen.add(
          key
        );
      }
    }


    return true;
  }

    /* ============================================================
     AUDIT LOG

     The audit is shared by generation, preflight and deployment.
  ============================================================ */

  /* ============================================================
     FRESH ROUTE VERIFICATION FOR ONE ROTATION

     This is the multi-hub equivalent of the proven v0.5.1
     verifyFreshRoutes().

     Each requested route is resolved again from the fresh
     planning dataset belonging to its own hub.

     We verify:

       permanent line ID
       distance
       airport category
       aircraft category
       aircraft range
       duration

     The same-type AM lineList duration remains authoritative
     when available.
  ============================================================ */

  /**
   * verifyFreshRoutes()
   * Rechecks requested routes against newly fetched planning data before proceeding.
   *
   * Notes: This describes the existing behaviour; no logic is changed.
   */
  async function verifyFreshRoutes(
    generated,
    freshCatalogues,
    freshAircraftSpec,
    freshTypeCatalogue,
    audit = null
  ) {

    const verifiedRoutes =
      [];


    if (audit) {

      auditInfo(
        audit,
        `${generated.rotationName}: re-checking ` +
        `${generated.destinations.length} requested routes`
      );
    }


for (
  let routeIndex = 0;
  routeIndex <
    generated.destinations.length;
  routeIndex++
) {

  const destination =
    generated.destinations[
      routeIndex
    ];


  const destinationCode =
    normaliseCode(
      destination
    );


      const freshRoute =
        resolveRoute(
          freshCatalogues,
          generated.aircraftType,
          generated.hub,
          destinationCode,
          freshTypeCatalogue
        );


      if (!freshRoute) {

        throw new Error(
          `${generated.rotationName}: could not find route ` +
          `generated.hub/{destinationCode} in fresh ` +
          `${generated.hub} planning data.`
        );
      }


      /*
       * Match the freshly resolved route to the exact route
       * stored during Generate Batch.
       */

const originalRoute =
  generated.resolvedRoutes[
    routeIndex
  ];


if (!originalRoute) {

  throw new Error(
    `${generated.rotationName}: generated route occurrence ` +
    `routeIndex+1({generated.hub}/${destinationCode}) ` +
    `is missing from the locked batch.`
  );
}


if (
  normaliseCode(
    originalRoute.destination
  ) !==
  destinationCode
) {

  throw new Error(
    `${generated.rotationName}: route occurrence ` +
    `${routeIndex + 1} changed from ` +
    `generated.hub/{normaliseCode(
      originalRoute.destination
    )} to ` +
    `generated.hub/{destinationCode}. ` +
    `Generate the batch again.`
  );
}


      /* ========================================================
         LINE ID
      ======================================================== */

      if (
        Number(
          freshRoute.id
        ) !==
        Number(
          originalRoute.id
        )
      ) {

        throw new Error(
          `${generated.rotationName}: ${generated.hub}/` +
          `${destinationCode} line ID changed from ` +
          `${originalRoute.id} to ${freshRoute.id}. ` +
          `Generate the batch again.`
        );
      }


      /* ========================================================
         DISTANCE
      ======================================================== */

      const freshDistance =
        Number(
          freshRoute.distance
        );


      const originalDistance =
        Number(
          originalRoute.distance
        );


      if (
        !Number.isFinite(
          freshDistance
        ) ||
        freshDistance <=
          0
      ) {

        throw new Error(
          `${generated.rotationName}: ${generated.hub}/` +
          `${destinationCode} has an invalid fresh distance.`
        );
      }


      if (
        freshDistance !==
        originalDistance
      ) {

        throw new Error(
          `${generated.rotationName}: ${generated.hub}/` +
          `${destinationCode} distance changed from ` +
          `${originalDistance.toLocaleString()} km to ` +
          `${freshDistance.toLocaleString()} km. ` +
          `Generate the batch again.`
        );
      }


      /* ========================================================
         AIRCRAFT RANGE
      ======================================================== */

      const aircraftRange =
        Number(
          freshAircraftSpec.range
        );


      if (
        !Number.isFinite(
          aircraftRange
        ) ||
        aircraftRange <=
          0
      ) {

        throw new Error(
          `${generated.rotationName}: ${generated.aircraftType} ` +
          `has no valid fresh aircraft range.`
        );
      }


      if (
        freshDistance >
        aircraftRange
      ) {

        throw new Error(
          `${generated.rotationName}: ${generated.hub}/` +
          `${destinationCode} is ` +
          `${freshDistance.toLocaleString()} km, exceeding ` +
          `${generated.aircraftType} range of ` +
          `${aircraftRange.toLocaleString()} km.`
        );
      }


      /* ========================================================
         AIRPORT CATEGORY
      ======================================================== */

      const freshRouteCategory =
        Number(
          freshRoute.category
        );


      const originalRouteCategory =
        Number(
          originalRoute.category
        );


      const aircraftCategory =
        Number(
          freshAircraftSpec.category
        );


      if (
        !Number.isFinite(
          freshRouteCategory
        )
      ) {

        throw new Error(
          `${generated.rotationName}: ${generated.hub}/` +
          `${destinationCode} has no valid fresh airport category.`
        );
      }


      if (
        !Number.isFinite(
          aircraftCategory
        )
      ) {

        throw new Error(
          `${generated.rotationName}: ${generated.aircraftType} ` +
          `has no valid aircraft category.`
        );
      }


      if (
        freshRouteCategory <
        aircraftCategory
      ) {

        throw new Error(
          `${generated.rotationName}: ${generated.hub}/` +
          `${destinationCode} is airport category ` +
          `${freshRouteCategory}, but ` +
          `${generated.aircraftType} requires category ` +
          `${aircraftCategory} or higher.`
        );
      }


      if (
        !Number.isFinite(
          originalRouteCategory
        )
      ) {

        throw new Error(
          `${generated.rotationName}: generated route ` +
          `generated.hub/{destinationCode} has no locked ` +
          `airport category. Generate again.`
        );
      }


      if (
        freshRouteCategory !==
        originalRouteCategory
      ) {

        throw new Error(
          `${generated.rotationName}: ${generated.hub}/` +
          `${destinationCode} airport category changed from ` +
          `${originalRouteCategory} to ${freshRouteCategory}. ` +
          `Generate the batch again.`
        );
      }


      /* ========================================================
         DURATION
      ======================================================== */

      const freshDuration =
        Number(
          freshRoute.duration
        );


      const originalDuration =
        Number(
          originalRoute.duration
        );


      if (
        !Number.isFinite(
          freshDuration
        ) ||
        freshDuration <=
          0
      ) {

        throw new Error(
          `${generated.rotationName}: ${generated.hub}/` +
          `${destinationCode} produced an invalid fresh duration.`
        );
      }


      if (
        freshDuration !==
        originalDuration
      ) {

        throw new Error(
          `${generated.rotationName}: ${generated.hub}/` +
          `${destinationCode} duration changed from ` +
          `${secondsToDuration(
            originalDuration
          )} to ${secondsToDuration(
            freshDuration
          )}. Generate the batch again.`
        );
      }


      /*
       * Final shared range/category guard.
       */

      validateRange(
        freshRoute,
        freshAircraftSpec
      );


      if (audit) {

        auditOk(
          audit,
          `${generated.rotationName}: ` +
          `generated.hub/{destinationCode} · ` +
          `${freshDistance.toLocaleString()} km · ` +
          `category ${freshRouteCategory} · ` +
          `${secondsToDuration(
            freshDuration
          )} · ${freshRoute.durationSource}`
        );
      }


      verifiedRoutes.push(
        freshRoute
      );
    }


    if (
      verifiedRoutes.length !==
      generated.destinations.length
    ) {

      throw new Error(
        `${generated.rotationName}: fresh route verification ` +
        `returned ${verifiedRoutes.length} routes; expected ` +
        `${generated.destinations.length}.`
      );
    }


    if (audit) {

      auditOk(
        audit,
        `${generated.rotationName}: ` +
        `verifiedRoutes.length/{generated.destinations.length} ` +
        `routes freshly verified`
      );
    }


    return verifiedRoutes;
  }


  /* ============================================================
     VALIDATE ONE ROTATION FROM ONE FRESH HUB DATASET

     This does not perform another network request.

     The batch preflight loads each hub once and passes the same
     fresh hub response to every rotation belonging to that hub.
  ============================================================ */

  /**
   * validateFreshRotation()
   * Validates one generated rotation against a fresh live snapshot.
   *
   * Notes: This describes the existing behaviour; no logic is changed.
   */
  async function validateFreshRotation(
    generated,
    freshHubEntry,
    audit = null
  ) {

    if (!generated) {

      throw new Error(
        "No generated rotation is available."
      );
    }


    if (!freshHubEntry) {

      throw new Error(
        `${generated.rotationName}: fresh hub data is unavailable.`
      );
    }


    const expectedFingerprint =
      makeRotationFingerprint(
        generated
      );


    if (
      normaliseCode(
        freshHubEntry.code
      ) !==
      normaliseCode(
        generated.hub
      )
    ) {

      throw new Error(
        `${generated.rotationName}: expected fresh ` +
        `${generated.hub} data but received ${freshHubEntry.code}.`
      );
    }


    if (
      Number(
        freshHubEntry.hubId
      ) !==
      Number(
        generated.hubId
      )
    ) {

      throw new Error(
        `${generated.rotationName}: fresh hub ID changed from ` +
        `${generated.hubId} to ${freshHubEntry.hubId}.`
      );
    }


    const freshData =
      freshHubEntry.data;


    const freshTypeCatalogue =
      freshHubEntry.typeCatalogue;


    const freshCatalogues =
      freshHubEntry.routeCatalogues;


    /* ==========================================================
       EXACT SEVEN AIRCRAFT
    ========================================================== */

    const liveLockedAircraft =
      verifyFreshRotationAircraft(
        generated,
        freshData,
        audit
      );


    /* ==========================================================
       FRESH AIRCRAFT SPECIFICATION
    ========================================================== */

    const freshAircraftSpec =
      getAircraftTypeSpec(
        generated.aircraftType,
        freshTypeCatalogue
      );


    if (audit) {

      auditOk(
        audit,
        `${generated.rotationName}: ${generated.aircraftType} · ` +
        `${Number(
          freshAircraftSpec.speed
        ).toLocaleString()} km/h · ` +
        `${Number(
          freshAircraftSpec.range
        ).toLocaleString()} km range · ` +
        `category ${freshAircraftSpec.category}`
      );
    }


    /*
     * Aircraft specification itself must not have changed
     * since generation.
     */

    if (
      Number(
        freshAircraftSpec.speed
      ) !==
        Number(
          generated.aircraftSpec.speed
        ) ||
      Number(
        freshAircraftSpec.range
      ) !==
        Number(
          generated.aircraftSpec.range
        ) ||
      Number(
        freshAircraftSpec.category
      ) !==
        Number(
          generated.aircraftSpec.category
        )
    ) {

      throw new Error(
        `${generated.rotationName}: ${generated.aircraftType} ` +
        `specifications changed after generation. Generate again.`
      );
    }


    /* ==========================================================
       REQUESTED ROUTES
    ========================================================== */

    const verifiedRoutes =
      await verifyFreshRoutes(
        generated,
        freshCatalogues,
        freshAircraftSpec,
        freshTypeCatalogue,
        audit
      );


    /* ==========================================================
       REBUILD BASE PATTERN
    ========================================================== */

    const freshBasePattern =
      buildBasePattern(
        generated.hub,
        generated.destinations,
        generated.aircraftType,
        freshCatalogues,
        freshTypeCatalogue
      );


    if (
      Number(
        freshBasePattern.totalDuration
      ) !==
      Number(
        generated.basePattern.totalDuration
      )
    ) {

      throw new Error(
        `${generated.rotationName}: base pattern duration changed ` +
        `from ${secondsToDuration(
          generated.basePattern.totalDuration
        )} to ${secondsToDuration(
          freshBasePattern.totalDuration
        )}. Generate again.`
      );
    }


    if (audit) {

      auditOk(
        audit,
        `${generated.rotationName}: base pattern still ` +
        `${secondsToDuration(
          freshBasePattern.totalDuration
        )}`
      );
    }


    /*
     * Defensive per-route range/category validation using the
     * fresh aircraft specification.
     */

    for (
      const item of
      freshBasePattern.schedule
    ) {

      validateRange(
        item.route,
        freshAircraftSpec
      );
    }


    if (audit) {

      auditOk(
        audit,
        `${generated.rotationName}: all route distances and ` +
        `airport categories compatible`
      );
    }


    /* ==========================================================
       REBUILD SEVEN SHIFTED SCHEDULES
    ========================================================== */

    const freshRotations =
      buildAircraftRotations(
        liveLockedAircraft,
        freshBasePattern,
        generated.rotationName
      );


    if (
      freshRotations.length !==
      generated.rotations.length
    ) {

      throw new Error(
        `${generated.rotationName}: fresh aircraft rotation ` +
        `count changed unexpectedly.`
      );
    }


    /*
     * Exact permanent IDs and exact ordering must remain fixed.
     */

    for (
      let i = 0;
      i < freshRotations.length;
      i++
    ) {

      if (
        Number(
          freshRotations[i].aircraftId
        ) !==
        Number(
          generated.rotations[i].aircraftId
        )
      ) {

        throw new Error(
          `${generated.rotationName}: aircraft ordering changed ` +
          `unexpectedly at position ${i + 1}.`
        );
      }


      if (
        String(
          freshRotations[i].oldName
        ) !==
        String(
          generated.rotations[i].oldName
        )
      ) {

        throw new Error(
          `${generated.rotationName}: original aircraft name changed ` +
          `for permanent ID ${freshRotations[i].aircraftId}.`
        );
      }


      if (
        String(
          freshRotations[i].newName
        ) !==
        String(
          generated.rotations[i].newName
        )
      ) {

        throw new Error(
          `${generated.rotationName}: generated aircraft name changed ` +
          `for permanent ID ${freshRotations[i].aircraftId}.`
        );
      }
    }


    /* ==========================================================
       CYCLIC WEEKLY OVERLAPS
    ========================================================== */

    validateAllRotationSchedules(
      freshRotations
    );


    if (audit) {

      auditOk(
        audit,
        `${generated.rotationName}: no cyclic weekly overlaps`
      );
    }


    /* ==========================================================
       FLIGHT COUNT
    ========================================================== */

    const expectedFlights =
      7 *
      generated.destinations.length;


    const actualFlights =
      freshRotations.reduce(
        (
          total,
          aircraftRotation
        ) =>
          total +
          aircraftRotation.schedule.length,
        0
      );


    if (
      actualFlights !==
      expectedFlights
    ) {

      throw new Error(
        `${generated.rotationName}: expected ${expectedFlights} ` +
        `schedule entries but validated ${actualFlights}.`
      );
    }


    if (audit) {

      auditOk(
        audit,
        `${generated.rotationName}: ${actualFlights} schedule ` +
        `entries validated`
      );
    }


    /* ==========================================================
       EXACT NATIVE PAYLOADS MUST BE UNCHANGED
    ========================================================== */

    for (
      let i = 0;
      i < freshRotations.length;
      i++
    ) {

      const generatedPayload =
        canonicalPayload(
          generated.rotations[
            i
          ].payload
        );


      const freshPayload =
        canonicalPayload(
          freshRotations[
            i
          ].payload
        );


      if (
        JSON.stringify(
          generatedPayload
        ) !==
        JSON.stringify(
          freshPayload
        )
      ) {

        throw new Error(
          `${generated.rotationName}: native schedule payload for ` +
          `aircraft ID ${freshRotations[i].aircraftId} changed ` +
          `after generation. Generate again.`
        );
      }
    }


    if (audit) {

      auditOk(
        audit,
        `${generated.rotationName}: all 7 native schedule ` +
        `payloads unchanged`
      );
    }


    /* ==========================================================
       ROTATION FINGERPRINT
    ========================================================== */

    const finalFingerprint =
      makeRotationFingerprint(
        generated
      );


    if (
      finalFingerprint !==
      expectedFingerprint
    ) {

      throw new Error(
        `${generated.rotationName}: generated rotation changed ` +
        `during validation.`
      );
    }


    return {

      generated,

      freshData,

      freshTypeCatalogue,

      freshCatalogues,

      freshAircraftSpec,

      verifiedRoutes,

      liveLockedAircraft,

      freshBasePattern,

      freshRotations,

      totalFlights:
        actualFlights,

      fingerprint:
        finalFingerprint
    };
  }


  /* ============================================================
     CREATE FRESH HUB SNAPSHOTS FOR A BATCH

     Unlike refreshRequiredHubs(), this does NOT mutate the
     global hubRegistry.

     That is useful during preflight because we want one
     internally consistent snapshot per hub.
  ============================================================ */

  /**
   * loadFreshBatchHubSnapshots()
   * Loads current planning snapshots for the hubs referenced by the batch.
   *
   * Notes: This describes the existing behaviour; no logic is changed.
   */
  async function loadFreshBatchHubSnapshots(
    batch,
    audit = null
  ) {

    const snapshots =
      new Map();


    for (
      const hubCode of
      batch.hubs
    ) {

      const lockedEntry =
        requireHubEntry(
          hubCode
        );


      if (audit) {

        auditInfo(
          audit,
          `${hubCode}: loading fresh planning snapshot`
        );
      }


      const data =
        await fetchPlanningUrl(
          lockedEntry.planningUrl,
          lockedEntry.hubId
        );


      const resolvedCode =
        discoverHubCodeFromData(
          data
        );


      if (
        normaliseCode(
          resolvedCode
        ) !==
        normaliseCode(
          hubCode
        )
      ) {

        throw new Error(
          `${hubCode}: fresh planning URL resolved as ` +
          `${resolvedCode}.`
        );
      }


      const snapshot = {

        code:
          hubCode,

        hubId:
          Number(
            lockedEntry.hubId
          ),

        planningUrl:
          lockedEntry.planningUrl,

        data,

        typeCatalogue:
          buildAircraftTypeCatalogue(
            data
          ),

        routeCatalogues:
          buildRouteCatalogues(
            data
          ),

        loadedAt:
          Date.now()
      };


      snapshots.set(
        hubCode,
        snapshot
      );


      if (audit) {

        auditOk(
          audit,
          `${hubCode}: fresh planning snapshot loaded`
        );
      }
    }


    return snapshots;
  }


  /* ============================================================
     COMPLETE READ-ONLY BATCH VALIDATION

     This is the core v1.0 safety gate.

     No POST occurs anywhere in this function.
  ============================================================ */

  /**
   * validateFreshBatch()
   * Runs batch-wide live-data validation against fresh hub snapshots.
   *
   * Notes: This describes the existing behaviour; no logic is changed.
   */
  async function validateFreshBatch(
    batch,
    audit = null
  ) {

    if (!batch) {

      throw new Error(
        "No generated batch is available."
      );
    }


    /*
     * Validate our generated structure before performing any
     * fresh network work.
     */

    const structural =
      validateGeneratedBatch(
        batch
      );


    const expectedFingerprint =
      makeBatchFingerprint(
        batch
      );


    if (audit) {

      auditOk(
        audit,
        `${structural.totalRotations} rotation` +
        `${structural.totalRotations === 1 ? "" : "s"} · ` +
        `${structural.totalAircraft} aircraft · ` +
        `${structural.totalFlights} weekly flights in locked batch`
      );
    }


    /* ==========================================================
       LOAD ONE FRESH SNAPSHOT PER HUB
    ========================================================== */

    const freshHubs =
      await loadFreshBatchHubSnapshots(
        batch,
        audit
      );


    /* ==========================================================
       VALIDATE EVERY ROTATION AGAINST ITS OWN HUB SNAPSHOT
    ========================================================== */

    const freshRotationResults =
      [];


    for (
      let index = 0;
      index < batch.rotations.length;
      index++
    ) {

      const generated =
        batch.rotations[
          index
        ];


      const freshHubEntry =
        freshHubs.get(
          generated.hub
        );


      if (!freshHubEntry) {

        throw new Error(
          `${generated.rotationName}: no fresh ${generated.hub} ` +
          `hub snapshot is available.`
        );
      }


      if (audit) {

        auditInfo(
          audit,
          `Validating rotation ${index + 1}/` +
          `${batch.rotations.length}: ` +
          `${generated.rotationName}`
        );
      }


      const result =
        await validateFreshRotation(
          generated,
          freshHubEntry,
          audit
        );


      freshRotationResults.push(
        result
      );


      if (audit) {

        auditOk(
          audit,
          `${generated.rotationName}: fresh rotation validation passed`
        );
      }
    }


    /* ==========================================================
       WHOLE-BATCH AIRCRAFT UNIQUENESS
    ========================================================== */

    validateFreshAircraftCollections(
      freshRotationResults
    );


    const freshAircraftIds =
      freshRotationResults.flatMap(
        result =>
          result.liveLockedAircraft.map(
            aircraft =>
              Number(
                aircraft.id
              )
          )
      );


    if (
      freshAircraftIds.length !==
      batch.totalAircraft
    ) {

      throw new Error(
        `Fresh validation found ${freshAircraftIds.length} aircraft; ` +
        `expected ${batch.totalAircraft}.`
      );
    }


    if (
      new Set(
        freshAircraftIds
      ).size !==
      freshAircraftIds.length
    ) {

      throw new Error(
        "Fresh batch validation found duplicate permanent aircraft IDs."
      );
    }


    if (audit) {

      auditOk(
        audit,
`${freshAircraftIds.length}/${batch.totalAircraft} ` +
          `permanent aircraft IDs are unique across the whole batch`
      );
    }


    /* ==========================================================
       WHOLE-BATCH FLIGHT COUNT
    ========================================================== */

    const totalFlights =
      freshRotationResults.reduce(
        (
          total,
          result
        ) =>
          total +
          result.totalFlights,
        0
      );


    if (
      totalFlights !==
      batch.totalFlights
    ) {

      throw new Error(
        `Fresh batch contains ${totalFlights} weekly flights; ` +
        `expected ${batch.totalFlights}.`
      );
    }


    if (audit) {

      auditOk(
        audit,
`${totalFlights}/${batch.totalFlights} weekly schedule ` +
          `entries validated across the whole batch`
      );
    }


    /* ==========================================================
       ENSURE THE GENERATED BATCH ITSELF DID NOT CHANGE
    ========================================================== */

    assertBatchFingerprint(
      batch,
      expectedFingerprint
    );


    if (audit) {

      auditOk(
        audit,
        "Generated batch fingerprint unchanged"
      );
    }


    return {

      batch,

      freshHubs,

      freshRotationResults,

      totalRotations:
        batch.totalRotations,

      totalAircraft:
        batch.totalAircraft,

      totalFlights,

      fingerprint:
        expectedFingerprint
    };
  }






/* ============================================================
   VALIDATION DISPATCHER — v1.0

   Normal batches use validateFreshBatch().

   Recovery-mode batches are explicitly rejected.

   No OPT002 recovery validation or deployment is permitted.
============================================================ */

/**
 * validateFreshBatchForMode()
 * Selects the required live validation checks for the requested operating mode.
 *
 * Notes: This describes the existing behaviour; no logic is changed.
 */
async function validateFreshBatchForMode(
  batch,
  audit = null
) {

  // v1.0 — OPT002 recovery is disabled.
  // Reject recovery batches rather than processing them.

  if (
    batch?.recoveryMode === true
  ) {

    throw new Error(
      "OPT002 recovery is disabled in v1.0"
    );
  }

  // Normal batch preflight only.
  return await validateFreshBatch(
    batch,
    audit
  );
}



  /* ============================================================
     PREFLIGHT STATE

     Deploy is armed only if:

       - a batch exists
       - preflight passed
       - preflight belongs to that exact batch fingerprint
  ============================================================ */

  /**
   * batchDeploymentIsArmed()
   * Reports whether a valid preflight has unlocked the live deployment path.
   *
   * Notes: This describes the existing behaviour; no logic is changed.
   */
  function batchDeploymentIsArmed() {

    if (
      !generatedBatch ||
      !batchPreflightState ||
      !batchPreflightState.passed
    ) {

      return false;
    }


    try {

      return (
        batchPreflightState.fingerprint ===
        makeBatchFingerprint(
          generatedBatch
        )
      );

    } catch (error) {

      return false;
    }
  }


  /**
   * invalidateBatchPreflight()
   * Clears prior preflight authorisation when relevant state changes.
   *
   * Notes: This describes the existing behaviour; no logic is changed.
   */
  function invalidateBatchPreflight() {

    batchPreflightState =
      null;


    updateDeployButton();
  }


  /* ============================================================
     RUN LIVE BATCH PREFLIGHT

     READ ONLY.

     This is the button action used after Generate Batch.
  ============================================================ */

  /**
   * runLiveBatchPreflight()
   * Runs fresh read-only safety checks and arms deployment only if the exact batch passes.
   *
   * Notes: This describes the existing behaviour; no logic is changed.
   */
  async function runLiveBatchPreflight() {

    if (busy) {

      return;
    }


    if (!generatedBatch) {

      setStatus(
        "Generate a batch before running Live Batch Preflight.",

      );


      return;
    }


    const audit =

        "Live Batch Preflight"
      );


    setCurrentAudit(

    );


    invalidateBatchPreflight();


    setBusy(
      true
    );


    setStatus(
      "Running read-only live batch preflight…",
      "working"
    );


    try {

      auditInfo(
        audit,
        "LIVE BATCH PREFLIGHT — READ ONLY"
      );


      auditInfo(
        audit,
        `${generatedBatch.totalRotations} rotation` +
        `${generatedBatch.totalRotations === 1 ? "" : "s"} · ` +
        `${generatedBatch.totalAircraft} aircraft · ` +
        `${generatedBatch.totalFlights} weekly flights`
      );


      auditInfo(
        audit,
        `Hubs: ${generatedBatch.hubs.join(", ")}`
      );


const result =
  await validateFreshBatchForMode(
    generatedBatch,
    audit
  );

if (
  result.fingerprint !==
  makeBatchFingerprint(generatedBatch)
) {
  throw new Error(
    "Generated batch changed during preflight. " +
    "Generate and preflight again."
  );
}

batchPreflightState = {

        passed:
          true,

        checkedAt:
          Date.now(),

        fingerprint:
          result.fingerprint,

        totalRotations:
          result.totalRotations,

        totalAircraft:
          result.totalAircraft,

        totalFlights:
          result.totalFlights,

        hubs:
          [
            ...generatedBatch.hubs
          ]
      };


      auditOk(
        audit,
        "LIVE BATCH PREFLIGHT PASSED"
      );


      auditOk(
        audit,
        "Ready for live batch deployment"
      );


      setStatus(
        `Preflight passed: ${result.totalRotations} rotation` +
        `${result.totalRotations === 1 ? "" : "s"}, ` +
        `${result.totalAircraft} aircraft and ` +
        `${result.totalFlights} weekly flights are ready for deployment.`,
        "success"
      );


    } catch (error) {

      batchPreflightState =
        null;


      auditFail(
        audit,
        error?.message ||
        String(
          error
        )
      );


      auditFail(
        audit,
        "LIVE BATCH PREFLIGHT FAILED — no live changes were made"
      );


      setStatus(
        error?.message ||
        String(
          error
        ),
        "error"
      );


      console.error(
        "[AM Rotation Builder] Batch preflight failed:",
        error
      );

    } finally {

      setBusy(
        false
      );


      updateDeployButton();
    }
  }


  /* ============================================================
     END CHUNK 4
  ============================================================ */

   /* ============================================================
     DEPLOYMENT ERROR TYPE

     An ambiguous write means:

       - the POST was attempted exactly once
       - we cannot prove the requested state was applied

     Deployment MUST stop immediately.

     We never repeat the POST automatically.
  ============================================================ */

  class AmbiguousWriteError extends Error {

    constructor(
      message,
      details = null
    ) {

      super(
        message
      );


      this.name =
        "AmbiguousWriteError";


      this.ambiguous =
        true;


      this.details =
        details;
    }
  }


  /* ============================================================
     IMMEDIATE PRE-WRITE AIRCRAFT CHECK

     This deliberately performs a fresh GET immediately before
     scheduling each aircraft.

     The aircraft must still:

       exist
       be at the expected hub
       be the expected aircraft type
       have its original name
       be completely empty

     We never silently substitute another aircraft.
  ============================================================ */

  /**
   * assertAircraftStillReadyForWrite()
   * Checks an aircraft immediately before a live scheduling write to reduce stale-state risk.
   *
   * Notes: This describes the existing behaviour; no logic is changed.
   */
  async function assertAircraftStillReadyForWrite(
    generatedRotation,
    aircraftRotation
  ) {

    const freshData =
      await getFreshHubData(
        generatedRotation.hub
      );


    const aircraft =
      getAircraftFromPlanningData(
        freshData,
        aircraftRotation.aircraftId
      );


    if (!aircraft) {

      throw new Error(
        `${aircraftRotation.newName}: permanent aircraft ID ` +
        `${aircraftRotation.aircraftId} disappeared before write.`
      );
    }


    if (
      Number(
        aircraft.hubId
      ) !==
      Number(
        generatedRotation.hubId
      )
    ) {

      throw new Error(
        `${aircraftRotation.newName}: aircraft ID ` +
        `${aircraftRotation.aircraftId} is no longer at ` +
        `${generatedRotation.hub}.`
      );
    }


    if (
      String(
        aircraft.aircraftListName
      ) !==
      String(
        generatedRotation.aircraftType
      )
    ) {

      throw new Error(
        `${aircraftRotation.newName}: aircraft ID ` +
        `${aircraftRotation.aircraftId} is now ` +
        `${aircraft.aircraftListName}, not ` +
        `${generatedRotation.aircraftType}.`
      );
    }


    if (
      String(
        aircraft.name ??
        ""
      ) !==
      String(
        aircraftRotation.oldName
      )
    ) {

throw new Error(
  `${aircraftRotation.newName}: aircraft ID ` +
  `${aircraftRotation.aircraftId} is currently named ` +
  `"${aircraft.name}", but "${aircraftRotation.oldName}" ` +
  `was locked during generation.`
);
    }


    if (
      !isAircraftEmpty(
        aircraft
      )
    ) {

      throw new Error(
        `${aircraftRotation.newName}: aircraft ID ` +
        `${aircraftRotation.aircraftId} is no longer empty.`
      );
    }


    return {

      data:
        freshData,

      aircraft
    };
  }


  /* ============================================================
     SCHEDULE POST — EXACTLY ONCE

     Native Airlines Manager endpoint:

       POST /network/planning/0/ajax

     Form field:

       planningData = JSON.stringify(payload)

     IMPORTANT:

     This function never retries.
  ============================================================ */

  /**
   * postScheduleExactlyOnce()
   * Submits a native schedule write with single-attempt semantics to avoid accidental duplicate posts.
   *
   * Notes: This describes the existing behaviour; no logic is changed.
   */
  async function postScheduleExactlyOnce(
    payload
  ) {

    validateNativePayload(
      payload,
      payload.aircraftId
    );


    const body =
      new URLSearchParams();


    body.set(
      "planningData",
      JSON.stringify(
        payload
      )
    );


    let response =
      null;


    let responseText =
      "";


    let fetchError =
      null;


    try {

      response =
        await fetch(
          SCHEDULE_ENDPOINT,
          {
            method:
              "POST",

            credentials:
              "include",

            headers: {

              "Content-Type":
                "application/x-www-form-urlencoded; charset=UTF-8",

              "X-Requested-With":
                "XMLHttpRequest",

              "Accept":
                "application/json, text/javascript, */*; q=0.01"
            },

            body:
              body.toString()
          }
        );


      try {

        responseText =
          await response.text();

      } catch (error) {

        responseText =
          "";
      }


    } catch (error) {

      fetchError =
        error;
    }


    return {

      attempted:
        true,

      response,

      responseText,

      fetchError,

      ok:
        Boolean(
          response &&
          response.ok
        )
    };
  }


  /* ============================================================
     VERIFY SCHEDULE WITH ONE FRESH READBACK
  ============================================================ */

  /**
   * readBackAircraftSchedule()
   * Fetches the aircraft's current schedule after a write for verification.
   *
   * Notes: This describes the existing behaviour; no logic is changed.
   */
  async function readBackAircraftSchedule(
    generatedRotation,
    aircraftRotation
  ) {

    /*
     * Small fixed settling delay before ONE readback.
     *
     * This is not a retry and does not cause another POST.
     */

    await sleep(
      350
    );


    const freshData =
      await getFreshHubData(
        generatedRotation.hub
      );


    const verification =
      verifyAircraftScheduleInData(
        freshData,
        aircraftRotation
      );


    return {

      freshData,

      verification
    };
  }


  /* ============================================================
     SCHEDULE + VERIFY

     POST is attempted ONCE.

     Even if fetch throws or HTTP status is non-OK, we perform
     a fresh readback.

     If the exact desired schedule exists, the write is treated
     as verified.

     Otherwise the state is ambiguous and deployment stops.

     THERE IS NO AUTOMATIC RE-POST.
  ============================================================ */

  /**
   * submitAndVerifySchedule()
   * Writes one aircraft schedule and verifies the persisted result by reading it back.
   *
   * Notes: This describes the existing behaviour; no logic is changed.
   */
  async function submitAndVerifySchedule(
    generatedRotation,
    aircraftRotation,
    audit
  ) {

    auditInfo(
      audit,
      `${aircraftRotation.newName}: submitting native schedule once`
    );


    const postResult =
      await postScheduleExactlyOnce(
        aircraftRotation.payload
      );


    if (
      postResult.fetchError
    ) {

      auditWarn(
        audit,
        `${aircraftRotation.newName}: schedule POST did not return ` +
        `cleanly; checking live state before deciding`
      );

    } else if (
      !postResult.ok
    ) {

      auditWarn(
        audit,
        `${aircraftRotation.newName}: schedule POST returned HTTP ` +
        `${postResult.response?.status}; checking live state`
      );

    } else {

      auditOk(
        audit,
        `${aircraftRotation.newName}: schedule POST accepted`
      );
    }


    const readback =
      await readBackAircraftSchedule(
        generatedRotation,
        aircraftRotation
      );


    if (
      readback.verification.ok
    ) {

      auditOk(
        audit,
        `${aircraftRotation.newName}: exact schedule independently verified`
      );


      return {

        postResult,

        readback,

        verified:
          true
      };
    }


    throw new AmbiguousWriteError(
      `${aircraftRotation.newName}: schedule POST was attempted, ` +
      `but the exact requested schedule could not be verified. ` +
      `Deployment has stopped and the schedule will NOT be ` +
      `submitted again automatically.`,
      {
        aircraftId:
          aircraftRotation.aircraftId,

        expected:
          readback.verification.expected,

        actual:
          readback.verification.actual,

        postHttpStatus:
          postResult.response
            ?.status ??
          null,

        fetchError:
          postResult.fetchError
            ?.message ??
          null
      }
    );
  }


  /* ============================================================
     AIRCRAFT EDIT PAGE

     GET the edit page immediately before a rename and extract:

       aircraft[name]
       aircraft[_token]

     CSRF tokens are NEVER stored in generatedBatch.
  ============================================================ */

  /**
   * getAircraftEditState()
   * Retrieves the aircraft editing state needed for rename validation.
   *
   * Notes: This describes the existing behaviour; no logic is changed.
   */
  async function getAircraftEditState(
    aircraftId
  ) {

    const url =
      `/aircraft/edit/${Number(
        aircraftId
      )}`;


    const response =
      await fetch(
        url,
        {
          method:
            "GET",

          credentials:
            "include",

          headers: {

            "Accept":
              "text/html,application/xhtml+xml,application/xml;q=0.9,*/*;q=0.8"
          },

          cache:
            "no-store"
        }
      );


    if (!response.ok) {

      throw new Error(
        `Aircraft edit GET failed for ID ${aircraftId}: ` +
        `HTTP ${response.status}.`
      );
    }


    const html =
      await response.text();


    const documentObject =
      new DOMParser()
        .parseFromString(
          html,
          "text/html"
        );


    const tokenInput =
      documentObject.querySelector(
        'input[name="aircraft[_token]"]'
      );


    const nameInput =
      documentObject.querySelector(
        'input[name="aircraft[name]"]'
      );


    const token =
      String(
        tokenInput?.value ??
        ""
      );


    const name =
      String(
        nameInput?.value ??
        ""
      );


    if (!token) {

      throw new Error(
        `Aircraft ID ${aircraftId}: could not obtain a fresh ` +
        `rename CSRF token.`
      );
    }


    if (!nameInput) {

      throw new Error(
        `Aircraft ID ${aircraftId}: could not read the current ` +
        `aircraft name from the edit page.`
      );
    }


    return {

      aircraftId:
        Number(
          aircraftId
        ),

      url,

      token,

      name,

      html
    };
  }


  /* ============================================================

        token
      )
    );


    let response =
      null;


    let responseText =
      "";


    let fetchError =
      null;


    try {

      response =
        await fetch(
          url,
          {
            method:
              "POST",

            credentials:
              "include",

            headers: {

              "Content-Type":
                "application/x-www-form-urlencoded; charset=UTF-8",

              "X-Requested-With":
                "XMLHttpRequest",

              "Accept":
                "text/html, */*; q=0.01"
            },

            body:
              body.toString()
          }
        );


      try {

        responseText =
          await response.text();

      } catch (error) {

        responseText =

      }


    } catch (error) {

    return {

  /* ============================================================
     RENAME + VERIFY

     Before POST:
       GET fresh edit page
       verify old name
       obtain fresh CSRF token

     POST:
       exactly once

     After POST:
       GET edit page again
       verify exact desired name

     A failed verification is ambiguous and stops deployment.
  ============================================================ */

  /**
   * renameAndVerifyAircraft()
   * Renames an aircraft and confirms that the expected name was applied.
   *
   * Notes: This describes the existing behaviour; no logic is changed.
   */
  async function renameAndVerifyAircraft(
    aircraftRotation,
    audit
  ) {

    auditInfo(
      audit,
      `${aircraftRotation.newName}: retrieving fresh rename token`
    );


    const before =
      await getAircraftEditState(
        aircraftRotation.aircraftId
      );


    if (
      String(
        before.name
      ) !==
      String(
        aircraftRotation.oldName
      )
    ) {

      throw new Error(
        `${aircraftRotation.newName}: rename aborted because ` +
        `aircraft ID ${aircraftRotation.aircraftId} is currently ` +
        `"${before.name}", not expected original name ` +
        `"${aircraftRotation.oldName}".`
      );
    }


    auditInfo(
      audit,
      `${aircraftRotation.newName}: submitting rename once`
    );


    const postResult =
      await postRenameExactlyOnce(
        aircraftRotation.aircraftId,
        aircraftRotation.newName,
        before.token
      );


    if (
      postResult.fetchError
    ) {

      auditWarn(
        audit,
        `${aircraftRotation.newName}: rename POST did not return ` +
        `cleanly; checking live name`
      );

    } else if (
      !postResult.ok
    ) {

      auditWarn(
        audit,
        `${aircraftRotation.newName}: rename POST returned HTTP ` +
        `${postResult.response?.status}; checking live name`
      );

    } else {

      auditOk(
        audit,
        `${aircraftRotation.newName}: rename POST accepted`
      );
    }


    await sleep(
      250
    );


    const after =
      await getAircraftEditState(
        aircraftRotation.aircraftId
      );


    if (
      String(
        after.name
      ) ===
      String(
        aircraftRotation.newName
      )
    ) {

      auditOk(
        audit,
        `${aircraftRotation.newName}: new aircraft name independently verified`
      );


      return {

        postResult,

        before,

        after,

        verified:
          true
      };
    }


    throw new AmbiguousWriteError(
      `${aircraftRotation.newName}: rename POST was attempted, ` +
      `but the requested name could not be verified. ` +
      `Deployment has stopped and the rename will NOT be ` +
      `submitted again automatically.`,
      {
        aircraftId:
          aircraftRotation.aircraftId,

        oldName:
          aircraftRotation.oldName,

        desiredName:
          aircraftRotation.newName,

        observedName:
          after.name,

        postHttpStatus:
          postResult.response
            ?.status ??
          null,

        fetchError:
          postResult.fetchError
            ?.message ??
          null
      }
    );
  }


  /* ============================================================
     POST-RENAME SCHEDULE VERIFICATION

     Renaming should not alter the schedule.

     We prove that before moving to the next aircraft.
  ============================================================ */

  /**
   * verifyScheduleAfterRename()
   * Checks that the aircraft schedule still matches after the rename operation.
   *
   * Notes: This describes the existing behaviour; no logic is changed.
   */
  async function verifyScheduleAfterRename(
    generatedRotation,
    aircraftRotation,
    audit
  ) {

    const freshData =
      await getFreshHubData(
        generatedRotation.hub
      );


    const verification =
      verifyAircraftScheduleInData(
        freshData,
        aircraftRotation
      );


    if (
      !verification.ok
    ) {

      throw new Error(
        `${aircraftRotation.newName}: schedule no longer matches ` +
        `the requested schedule after rename. Deployment stopped.`
      );
    }


    auditOk(
      audit,
      `${aircraftRotation.newName}: schedule still exact after rename`
    );


    return verification;
  }




  /* ============================================================
     DEPLOY ONE AIRCRAFT

     ORDER IS DELIBERATE:

       1. immediate fresh empty-aircraft check
       2. schedule POST exactly once
       3. exact schedule readback
       4. rename GET + fresh CSRF
       5. rename POST exactly once
       6. exact name readback
       7. exact schedule readback again

     We schedule BEFORE rename.

     Therefore if rename fails, the aircraft is at least left
     with the correct schedule and its original name.
  ============================================================ */

  /**
   * deployOneAircraft()
   * Coordinates scheduling, verification and naming for one selected aircraft.
   *
   * Notes: This describes the existing behaviour; no logic is changed.
   */
  async function deployOneAircraft(
    generatedRotation,
    aircraftRotation,
    audit
  ) {

    auditInfo(
      audit,
      `${aircraftRotation.newName}: immediate pre-write safety check`
    );


    await assertAircraftStillReadyForWrite(
      generatedRotation,
      aircraftRotation
    );


    auditOk(
      audit,
      `${aircraftRotation.newName}: aircraft still empty, unchanged, ` +
      `at ${generatedRotation.hub}, and ready for write`
    );



/*
 * v1.0 — Final pre-scheduling safety check.
 *
 * No aircraft reconfiguration is performed.
 *
 * Confirm the aircraft is still empty, at the correct hub,
 * has the correct permanent ID, aircraft type and original
 * name immediately before scheduling.
 */
    await assertAircraftStillReadyForWrite(
      generatedRotation,
      aircraftRotation
    );

    const scheduleResult =
      await submitAndVerifySchedule(
        generatedRotation,
        aircraftRotation,
        audit
      );


    const renameResult =
      await renameAndVerifyAircraft(
        aircraftRotation,
        audit
      );


    const finalScheduleVerification =
      await verifyScheduleAfterRename(
        generatedRotation,
        aircraftRotation,
        audit
      );

    return {

      aircraftId:
        aircraftRotation.aircraftId,

      oldName:
        aircraftRotation.oldName,

      newName:
        aircraftRotation.newName,

      scheduleResult,

      renameResult,

      finalScheduleVerification,

      completed:
        true
    };
  }


  /* ============================================================
     DEPLOY ONE COMPLETE 7-AIRCRAFT ROTATION

     Sequential only.

     Never Promise.all().
  ============================================================ */

  /**
   * deployOneRotation()
   * Deploys the seven aircraft associated with one generated rotation.
   *
   * Notes: This describes the existing behaviour; no logic is changed.
   */
  async function deployOneRotation(
    generatedRotation,
    rotationNumber,
    rotationCount,
    audit,
    deploymentProgress
  ) {

auditInfo(
  audit,
  `ROTATION ${rotationNumber}/${rotationCount}: ` +
  `${generatedRotation.rotationName} · ` +
  `${generatedRotation.hub} · ` +
  `${generatedRotation.aircraftType}`
);

    const completedAircraft =
      [];


    for (
      let aircraftIndex = 0;
      aircraftIndex <
        generatedRotation.rotations.length;
      aircraftIndex++
    ) {

      const aircraftRotation =
        generatedRotation.rotations[
          aircraftIndex
        ];


      auditInfo(
        audit,
        `${generatedRotation.rotationName}: aircraft ` +
        `${aircraftIndex + 1}/7 · ID ` +
        `${aircraftRotation.aircraftId} · ` +
        `${aircraftRotation.oldName} → ${aircraftRotation.newName}`
      );


/*
 * v1.0 — Ordinary deployment only.
 *
 * OPT002 recovery mode is disabled.
 * Each aircraft must pass the normal fresh
 * ready-for-write checks before scheduling.
 */

if (
  generatedRotation.recoveryMode
) {

  throw new Error(
    "Recovery-mode deployment is disabled in v1.0"
  );
}


const result =
  await deployOneAircraft(
    generatedRotation,
    aircraftRotation,
    audit
  );

completedAircraft.push(
  result
);

deploymentProgress.completedAircraft++;

      deploymentProgress.completedFlights +=
        aircraftRotation.schedule.length;


      auditOk(
        audit,
        `${generatedRotation.rotationName}: aircraft ` +
        `${aircraftIndex + 1}/7 complete`
      );


    }
if (
  completedAircraft.length !== 7
) {
  throw new Error(
    `${generatedRotation.rotationName}: expected 7 completed aircraft, ` +
    `but recorded ${completedAircraft.length}.`
  );
}

    deploymentProgress.completedRotations++;


    auditOk(
      audit,
      `${generatedRotation.rotationName}: ` +
      `7/7 aircraft successfully deployed`
    );


    return {

      rotationName:
        generatedRotation.rotationName,

      hub:
        generatedRotation.hub,

      aircraftType:
        generatedRotation.aircraftType,

      completedAircraft,

      completed:
        true
    };
  }


/* ============================================================
   FINAL ALL-BATCH SCHEDULE READBACK — v1.0

   Runs after every requested aircraft has completed.

   Verifies every aircraft's exact schedule using fresh
   planning data.

   No aircraft seating or cargo configuration checks.
============================================================ */

/**
 * verifyAllDeployedSchedules()
 * Performs a final check of all schedules written during batch deployment.
 *
 * Notes: This describes the existing behaviour; no logic is changed.
 */
async function verifyAllDeployedSchedules(
  batch,
  audit
) {

  auditInfo(
    audit,
    "Running final whole-batch verification"
  );

  /*
   * One fresh planning GET per hub.
   */

  const freshByHub = new Map();

  for (const hub of batch.hubs) {

    const freshData =
      await getFreshHubData(hub);

    freshByHub.set(
      hub,
      freshData
    );

    auditOk(
      audit,
      `${hub}: final planning snapshot loaded`
    );
  }

  let verifiedAircraft = 0;
  let verifiedFlights = 0;

  /*
   * Verify every aircraft's exact schedule.
   */

  for (
    const generatedRotation of
    batch.rotations
  ) {

    const freshData =
      freshByHub.get(
        generatedRotation.hub
      );

    if (!freshData) {

      throw new Error(
        `${generatedRotation.hub}: final planning snapshot missing.`
      );
    }

    for (
      const aircraftRotation of
      generatedRotation.rotations
    ) {

      const verification =
        verifyAircraftScheduleInData(
          freshData,
          aircraftRotation
        );

      if (!verification.ok) {

        throw new Error(
          `${aircraftRotation.newName}: final whole-batch ` +
          `schedule verification failed.`
        );
      }

      auditOk(
        audit,
        `${aircraftRotation.newName}: final schedule verified`
      );

      verifiedAircraft++;

      verifiedFlights +=
        aircraftRotation.schedule.length;
    }
  }

  /*
   * Verify final aircraft count.
   */

  if (
    verifiedAircraft !==
    batch.totalAircraft
  ) {

    throw new Error(
      `Final verification checked ${verifiedAircraft} aircraft; ` +
      `expected ${batch.totalAircraft}.`
    );
  }

  /*
   * Verify final flight count.
   */

  if (
    verifiedFlights !==
    batch.totalFlights
  ) {

    throw new Error(
      `Final verification checked ${verifiedFlights} flights; ` +
      `expected ${batch.totalFlights}.`
    );
  }

  auditOk(
    audit,
    `${verifiedAircraft}/${batch.totalAircraft} aircraft ` +
    `verified in final whole-batch readback`
  );

  auditOk(
    audit,
    `${verifiedFlights}/${batch.totalFlights} weekly flights ` +
    `verified in final whole-batch readback`
  );

  return {
    verifiedAircraft,
    verifiedFlights
  };
}

  /* ============================================================
     LIVE BATCH DEPLOYMENT

     SAFETY SEQUENCE:

       passed Live Batch Preflight
              ↓
       user confirmation
              ↓
       COMPLETE fresh final batch validation
              ↓
       rotation 1 aircraft 1..7
              ↓
       rotation 2 aircraft 1..7
              ↓
       ...
              ↓
       final whole-batch schedule readback

     NO WRITE occurs before the final fresh validation passes.
  ============================================================ */

  /**
   * deployLiveBatch()
   * Controls the live deployment sequence, enforcing preflight and recording outcomes.
   *
   * Notes: This describes the existing behaviour; no logic is changed.
   */
  async function deployLiveBatch(
  confirmedFingerprint = null
) {

    if (busy) {

      return;
    }


    if (!generatedBatch) {

      setStatus(
        "Generate a batch before deployment.",
        "error"
      );


      return;
    }


    if (
      !batchDeploymentIsArmed()
    ) {

      setStatus(
        "Run Live Batch Preflight successfully before deployment.",
        "error"
      );


      return;
    }




    const audit =
      createAuditLog(
        "LIVE Batch Deployment"
      );


    setCurrentAudit(
      audit
    );


    setBusy(
      true
    );


    const deploymentProgress = {

      completedRotations:
        0,

      completedAircraft:
        0,

      completedFlights:
        0
    };


    try {

      auditWarn(
        audit,
        "LIVE DEPLOYMENT — writes are now permitted after final safety validation"
      );


      auditInfo(
        audit,
        `${generatedBatch.totalRotations} rotation` +
        `${generatedBatch.totalRotations === 1 ? "" : "s"} · ` +
        `${generatedBatch.totalAircraft} aircraft · ` +
        `${generatedBatch.totalFlights} weekly flights`
      );


      setStatus(
        "Running final fresh batch validation immediately before first write…",
        "working"
      );


      /* ========================================================
         FINAL PRE-WRITE SAFETY GATE

         Still read-only.

         If anything changed since the user's earlier preflight,
         deployment stops before the first POST.
      ======================================================== */

       const finalValidation =
        await validateFreshBatchForMode(
          generatedBatch,
          audit
        );

if (
  confirmedFingerprint !== null &&
  finalValidation.fingerprint !==
    confirmedFingerprint
) {

  throw new Error(
    "Final validation fingerprint differs from the confirmed batch."
  );
}

      auditOk(
        audit,
        "FINAL PRE-WRITE BATCH VALIDATION PASSED"
      );


      auditWarn(
        audit,
        "Beginning sequential live writes"
      );


      /* ========================================================
         SEQUENTIAL ROTATIONS
      ======================================================== */

      const completedRotationResults =
        [];


      for (
        let rotationIndex = 0;
        rotationIndex <
          generatedBatch.rotations.length;
        rotationIndex++
      ) {

        const generatedRotation =
          generatedBatch.rotations[
            rotationIndex
          ];


        setStatus(
          `Deploying rotation ${rotationIndex + 1}/` +
          `${generatedBatch.totalRotations}: ` +
          `${generatedRotation.rotationName}…`,
          "working"
        );


        const result =
          await deployOneRotation(
            generatedRotation,
            rotationIndex + 1,
            generatedBatch.totalRotations,
            audit,
            deploymentProgress
          );


        completedRotationResults.push(
          result
        );
      }


      /* ========================================================
         FINAL READBACK
      ======================================================== */

      setStatus(
        "All writes completed. Running final whole-batch verification…",
        "working"
      );


      const finalVerification =
        await verifyAllDeployedSchedules(
          generatedBatch,
          audit
        );


      auditOk(
        audit,
        "LIVE BATCH DEPLOYMENT COMPLETE"
      );


      auditOk(
        audit,
        `${deploymentProgress.completedRotations}/` +
        `${generatedBatch.totalRotations} rotations deployed`
      );


      auditOk(
        audit,
        `${deploymentProgress.completedAircraft}/` +
        `${generatedBatch.totalAircraft} aircraft deployed`
      );


      auditOk(
        audit,
        `${deploymentProgress.completedFlights}/` +
        `${generatedBatch.totalFlights} weekly flights deployed`
      );


      setStatus(
        `Deployment complete: ` +
        `${generatedBatch.totalRotations} rotation` +
        `${generatedBatch.totalRotations === 1 ? "" : "s"}, ` +
        `${generatedBatch.totalAircraft} aircraft and ` +
        `${generatedBatch.totalFlights} weekly flights successfully deployed.`,
        "success"
      );


      /*
       * The preflight must be invalidated after success because
       * these aircraft are no longer empty and their names have
       * changed.
       */

      batchPreflightState =
        null;


      return {

        completedRotationResults,

        finalVerification,

        deploymentProgress
      };


    } catch (error) {

      /*
       * NEVER automatically retry here.
       *
       * Some earlier aircraft may already have been successfully
       * changed. We report exactly how far deployment progressed.
       */

      batchPreflightState =
        null;


      const ambiguous =
        Boolean(
          error?.ambiguous
        );


      if (ambiguous) {

        auditFail(
          audit,
          `AMBIGUOUS WRITE — ${error.message}`
        );

      } else {

        auditFail(
          audit,
          error?.message ||
          String(
            error
          )
        );
      }


      auditFail(
        audit,
        `DEPLOYMENT STOPPED after ` +
        `${deploymentProgress.completedRotations}/` +
        `${generatedBatch.totalRotations} complete rotations, ` +
        `${deploymentProgress.completedAircraft}/` +
        `${generatedBatch.totalAircraft} complete aircraft and ` +
        `${deploymentProgress.completedFlights}/` +
        `${generatedBatch.totalFlights} verified flights.`
      );


      if (
        deploymentProgress.completedAircraft >
        0
      ) {

        auditWarn(
          audit,
          "Aircraft completed before the failure remain changed. " +
          "No automatic rollback has been attempted."
        );


        auditWarn(
          audit,
          "Do NOT press Deploy again until the live state has been reviewed."
        );
      }


      setStatus(
        ambiguous
          ? (
              "Deployment stopped on an ambiguous live write. " +
              "Do not redeploy automatically; review the audit and live aircraft state."
            )
          : (
              "Deployment stopped. Some earlier aircraft may already have been deployed; review the audit before taking further action."
            ),
        "error"
      );


      console.error(
        "[AM Rotation Builder] Live batch deployment stopped:",
        error
      );


    } finally {

      setBusy(
        false
      );


      updateDeployButton();
    }
  }

   /* ============================================================
     GENERATE BATCH

     READ ONLY.

     Flow:

       parse multiline request
              ↓
       ensure required hubs exist
              ↓
       refresh each required hub
              ↓
       validate aircraft capacity
              ↓
       reserve unique aircraft

     Generate NEVER arms deployment.

     Live Batch Preflight must still be run separately.

  /**
   * generateBatch()
   * Parses user input, refreshes hub data, validates routes and aircraft, and builds a read-only batch preview.
   *
   * Notes: This describes the existing behaviour; no logic is changed.
   */
  async function generateBatch() {

    if (busy) {

      return;
    }


    const input =
      document.getElementById(
        "amrb-batch-input"
      );


    if (!input) {

      setStatus(
        "Batch input control is unavailable.",
        "error"
      );


      return;
    }


    const raw =
      input.value;


    const audit =
      createAuditLog(
        "Generate Batch"
      );


    setCurrentAudit(
      audit
    );


    generatedBatch =
      null;


    batchPreflightState =
      null;


    updateDeployButton();


    setBusy(
      true
    );


    setStatus(
      "Generating batch…",
      "working"
    );


    try {

      /* ========================================================
         PARSE REQUEST
      ======================================================== */

      const requests =
        parseBatchInput(
          raw
        );


      auditOk(
        audit,
        `${requests.length} rotation request` +
        `${requests.length === 1 ? "" : "s"} parsed`
      );


      const requiredHubs =
        getRequiredHubCodes(
          requests
        );


      auditInfo(
        audit,
        `Required hubs: ${requiredHubs.join(", ")}`
      );


      /* ========================================================
         ENSURE REQUIRED HUBS ARE DISCOVERED

         First inspect everything the current Planning page has
         exposed.

         Chunk 7 adds the UI/discovery controls that make this
         available to the user.
      ======================================================== */

      await discoverAvailableHubs(
        4000
      );


      const missingHubs =
        requiredHubs.filter(
          hub =>
            !getHubEntry(
              hub
            )
        );


      if (
        missingHubs.length >
        0
      ) {

        throw new Error(
          `Could not discover planning data for ` +
          `${missingHubs.join(", ")}. ` +
          `Open those hubs in Airlines Manager Planning so their ` +
          `planning datasets are available, then run Generate Batch again.`
        );
      }


        auditOk(
            audit,
            `${requiredHubs.length} required hubs discovered`
        );

      /* ========================================================
         FRESH READ-ONLY HUB DATA FOR GENERATION

         Generation should not rely on stale data from when the
         page was first opened.
      ======================================================== */

      await refreshRequiredHubs(
        requests,
        audit
      );


      /* ========================================================
         BUILD COMPLETE BATCH
      ======================================================== */

      const batch =
        buildRotationBatch(
          requests
        );


      const structural =
        validateGeneratedBatch(
          batch
        );


      batch.fingerprint =
        makeBatchFingerprint(
          batch
        );


      generatedBatch =
        batch;


      /*
       * Generate never carries forward an earlier preflight.
       */

      batchPreflightState =
        null;


      auditOk(
        audit,
        `${structural.totalRotations} rotation` +
        `${structural.totalRotations === 1 ? "" : "s"} generated`
      );


      auditOk(
        audit,
        `${structural.totalAircraft} unique permanent aircraft IDs locked`
      );


      auditOk(
        audit,
        `${structural.totalFlights} weekly schedule entries generated`
      );


      auditOk(
        audit,
        "Generated batch passed structural validation"
      );


      renderGeneratedBatch(
        generatedBatch
      );


      setStatus(
        `Batch generated: ${batch.totalRotations} rotation` +
        `${batch.totalRotations === 1 ? "" : "s"}, ` +
        `${batch.totalAircraft} aircraft, ` +
        `${batch.totalFlights} weekly flights. ` +
        `Run Live Batch Preflight before deployment.`,
        "success"
      );


    } catch (error) {

      generatedBatch =
        null;


      batchPreflightState =
        null;


      auditFail(
        audit,
        error?.message ||
        String(
          error
        )
      );


      setStatus(
        error?.message ||
        String(
          error
        ),
        "error"
      );


      clearBatchPreview();


      console.error(
        "[AM Rotation Builder] Generate Batch failed:",
        error
      );


    } finally {

      setBusy(
        false
      );


      updateDeployButton();
    }
  }

  /**
   * formatPercentage()
   * Formats percentage values for readable output.
   *
   * Notes: This describes the existing behaviour; no logic is changed.
   */
  function formatPercentage(
    value
  ) {

    const numeric =
      Number(
        value
      );


    if (
      !Number.isFinite(
        numeric
      )
    ) {

      return "—";
    }


    return (
      numeric.toFixed(
        1
      ) +
      "%"
    );
  }


  /**
   * formatDistance()
   * Formats distance values for readable output.
   *
   * Notes: This describes the existing behaviour; no logic is changed.
   */
  function formatDistance(
    value
  ) {

    const numeric =
      Number(
        value
      );


    if (
      !Number.isFinite(
        numeric
      )
    ) {

      return "—";
    }


    return (
      numeric.toLocaleString() +
      " km"
    );
  }


  /**
   * durationSourceLabel()

  function durationSourceLabel(

    if (

    return "Formula";

  /**
   * getRotationFlightCount()
   * Calculates the number of flights represented by a rotation.
   *
   * Notes: This describes the existing behaviour; no logic is changed.
   */
  function getRotationFlightCount(
    generated
  ) {

    return generated.rotations.reduce(
      (
        total,
        aircraftRotation
      ) =>
        total +
        aircraftRotation.schedule.length,
      0
    );
  }

  /* ============================================================
     MANUAL READ-ONLY HUB DISCOVERY

     Useful if the user has navigated between hubs in the AM
     planning interface and wants the builder to rescan the
     resource list before generating a batch.
  ============================================================ */

  /**
   * runHubDiscovery()
   * Starts hub discovery from the interface and updates status and controls.
   *
   * Notes: This describes the existing behaviour; no logic is changed.
   */
  async function runHubDiscovery() {

    if (busy) {

      return;
    }


    const audit =
      createAuditLog(
        "Hub Discovery"
      );


    setCurrentAudit(
      audit
    );


    setBusy(
      true
    );


    setStatus(
      "Discovering Airlines Manager planning hubs…",
      "working"
    );


    try {

      const before =
        hubRegistry.size;


      const total =
        await discoverAvailableHubs(
          5000
        );


      renderHubRegistry();


      const discovered =
        [
          ...hubRegistry.keys()
        ]
          .sort(
            naturalCompare
          );


      auditOk(
        audit,
        `${total} planning hub` +
        `${total === 1 ? "" : "s"} currently registered`
      );


      if (
        discovered.length
      ) {

        auditInfo(
          audit,
          `Hubs: ${discovered.join(", ")}`
        );
      }


      const newlyAdded =
        Math.max(
          0,
          total -
          before
        );


      setStatus(
        `Hub discovery complete: ${total} hub` +
        `${total === 1 ? "" : "s"} registered` +
        `${newlyAdded ? `, ${newlyAdded} newly added` : ""}.`,
        "success"
      );


    } catch (error) {

      auditFail(
        audit,
        error?.message ||
        String(
          error
        )
      );


      setStatus(
        error?.message ||
        String(
          error
        ),
        "error"
      );


    } finally {

      setBusy(
        false
      );
    }
  }

  /**
   * renderAudit()
   * Renders timestamped audit entries in the user interface.
   *
   * Notes: This describes the existing behaviour; no logic is changed.
   */
  function renderAudit(
    audit = currentAudit
  ) {

    const container =
      document.getElementById(
        "amrb-audit"
      );


    if (!container) {

      return;
    }


    if (
      !audit ||
      !Array.isArray(
        audit.entries
      ) ||
      audit.entries.length ===
        0
    ) {

      container.innerHTML = `
        <div class="amrb-empty-audit">
          No audit activity yet.
        </div>
      `;


      return;
    }


    const entries =
      audit.entries
        .map(
          entry => {

            let icon =
              "•";


            if (
              entry.level ===
              "ok"
            ) {

              icon =
                "✓";
            }


            if (
              entry.level ===
              "fail"
            ) {

              icon =
                "✕";
            }


            if (
              entry.level ===
              "warn"
            ) {

              icon =
                "!";
            }


            const time =
              new Date(
                entry.time
              )
                .toLocaleTimeString();


            return `
              <div
                class="amrb-audit-row
                       amrb-audit-${escapeHtml(
                         entry.level
                       )}"
              >

                <span class="amrb-audit-icon">
                  ${escapeHtml(
                    icon
                  )}
                </span>

                <span class="amrb-audit-time">
                  ${escapeHtml(
                    time
                  )}
                </span>

                <span class="amrb-audit-message">
                  ${escapeHtml(
                    entry.message
                  )}
                </span>

              </div>
            `;
          }
        )
        .join("");


    container.innerHTML =
      entries;


    container.scrollTop =
      container.scrollHeight;
  }

        !armed;

  ============================================================ */

  /**
   * copyAuditToClipboard()
   * Copies the current audit output for debugging or recordkeeping.
   *
   * Notes: This describes the existing behaviour; no logic is changed.
   */
  async function copyAuditToClipboard() {

    if (
      !currentAudit ||
      !Array.isArray(
        currentAudit.entries
      ) ||
      currentAudit.entries.length ===
        0
    ) {

      setStatus(
        "There is no audit log to copy.",
        "info"
      );


      return;
    }


    const lines = [

      `${currentAudit.title}`,
      "",

      ...currentAudit.entries.map(
        entry => {

          const time =
            new Date(
              entry.time
            )
              .toLocaleTimeString();


          return (
            `[${time}] ` +
            `${String(
              entry.level
            ).toUpperCase()} ` +
            `${entry.message}`
          );
        }
      )
    ];


    try {

      await navigator.clipboard.writeText(
        lines.join(
          "\n"
        )
      );


      setStatus(
        "Audit log copied to clipboard.",
        "success"
      );


    } catch (error) {

      setStatus(
        "Could not copy the audit log automatically.",
        "error"
      );
    }
  }

  /**
   * initialHubDiscovery()
   * Performs initial discovery of available hubs during startup.
   *
   * Notes: This describes the existing behaviour; no logic is changed.
   */
  async function initialHubDiscovery() {

    const audit =
      createAuditLog(
        "Startup"
      );


    setCurrentAudit(
      audit
    );


    setStatus(
      "Discovering available planning hubs…",
      "working"
    );


    try {

      const total =
        await discoverAvailableHubs(
          7000
        );


      renderHubRegistry();


      if (
        total ===
        0
      ) {

        auditWarn(
          audit,
          "No planning datasets have been discovered yet."
        );


        setStatus(
          "No planning hubs discovered yet. Open Airlines Manager Planning for a hub, then use Discover / Refresh Hubs.",
          "info"
        );


        return;
      }


      const hubs =
        [
          ...hubRegistry.keys()
        ]
          .sort(
            naturalCompare
          );


      auditOk(
        audit,
        `${total} hub` +
        `${total === 1 ? "" : "s"} discovered`
      );


      auditInfo(
        audit,
        `Available: ${hubs.join(", ")}`
      );


      setStatus(
        `${total} planning hub` +
        `${total === 1 ? "" : "s"} discovered: ` +
        `${hubs.join(", ")}.`,
        "success"
      );


    } catch (error) {

      auditFail(
        audit,
        error?.message ||
        String(
          error
        )
      );


      setStatus(
        "Initial hub discovery failed. Use Discover / Refresh Hubs to try again.",
        "error"
      );

    } finally {

      updateDeployButton();
    }
  }

     PAGE CHECK

     The script is deliberately restricted to AM Planning.

     @match already does this, but fail closed if the script is

  ============================================================ */

  /**
   * isSupportedPlanningPage()
   * Determines whether the current URL is a supported planning page.
   *
   * Notes: This describes the existing behaviour; no logic is changed.
   */
  function isSupportedPlanningPage() {

    return (
      location.hostname ===
        "tycoon.airlines-manager.com" &&
      location.pathname.startsWith(
        "/network/planning"
      )
    );
  }


  /* ============================================================
     FATAL STARTUP DISPLAY

     If the userscript itself is incomplete, display a small
     error panel rather than leaving partially working live-write
     controls on screen.
  ============================================================ */

  /**
   * showFatalStartupError()
   * Reports a startup failure prominently so the user knows the builder did not initialise.
   *
   * Notes: This describes the existing behaviour; no logic is changed.
   */
  function showFatalStartupError(
    error
  ) {

    console.error(
      "[AM Rotation Builder] Fatal startup error:",
      error
    );


    const existing =
      document.getElementById(
        "amrb-panel"
      );


    if (existing) {

      existing.remove();
    }


    /*
     * Install styles if possible so the error is readable.
     */

    try {

      installStyles();

    } catch (styleError) {

      console.error(
        "[AM Rotation Builder] Could not install error styles:",
        styleError
      );
    }


    const panel =
      document.createElement(
        "div"
      );


    panel.id =
      "amrb-panel";


    panel.innerHTML = `

      <div class="amrb-header">

        <div class="amrb-header-left">

          <div class="amrb-title">
            AM Rotation Builder
          </div>

          <div class="amrb-version">
            v${escapeHtml(
              VERSION
            )}
            · startup blocked
          </div>

        </div>

      </div>


      <div class="amrb-panel-body">

        <div class="amrb-section">

          <div class="amrb-status amrb-status-error">

            The rotation builder failed its startup safety check.

          </div>


          <div
            class="amrb-format"
            style="margin-top:10px; white-space:pre-wrap;"
          >${escapeHtml(
            error?.message ||
            String(
              error
            )
          )}</div>


          <div class="amrb-help">

            Live deployment controls have not been enabled.

            Check that all eight v1.0 chunks were pasted in
            order and that none were truncated.

          </div>

        </div>

      </div>
    `;


    document.body.appendChild(
      panel
    );
  }

     No write endpoint is called during startup.
  ============================================================ */

  /**
   * startRotationBuilder()
   * Main entry point: checks the page, initialises the UI and starts the builder.
   *
   * Notes: This describes the existing behaviour; no logic is changed.
   */
  async function startRotationBuilder() {

    if (
      !isSupportedPlanningPage()
    ) {

      return;
    }


    try {

      const selfCheck =
        runBuilderSelfCheck();


      console.log(
        "[AM Rotation Builder] Self-check passed:",
        selfCheck
      );


      await initialiseBuilderUI();


      console.log(
        `[AM Rotation Builder] v${VERSION} ready`
      );


    } catch (error) {

      showFatalStartupError(
        error
      );
    }
  }



  /* ============================================================
     DOM READY
  ============================================================ */

  if (
    document.readyState ===
    "loading"
  ) {

    document.addEventListener(
      "DOMContentLoaded",
      () => {

        startRotationBuilder();

      },
      {
        once:
          true
      }
    );

  } else {

    startRotationBuilder();
  }
